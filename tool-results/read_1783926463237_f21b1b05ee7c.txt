    86→function parseTimeToMin(t: string): number {
    87→  const [h, m] = t.split(':').map(Number);
    88→  return (h || 0) * 60 + (m || 0);

    91→/** Convert any CSS color to an rgba string with given alpha. Falls back to the original color if parsing fails. */
    92→function toRgba(color: string, alpha: number): string {
    93→  if (!color) return `rgba(20, 184, 166, ${alpha})`;
    94→  const hex = color.trim().startsWith('#') ? color.trim().slice(1) : color.trim();
    95→  if (/^[0-9a-fA-F]{3}$/.test(hex)) {
    96→    const expanded = hex.split('').map((c) => c + c).join('');
    97→    const r = parseInt(expanded.slice(0, 2), 16);
    98→    const g = parseInt(expanded.slice(2, 4), 16);
    99→    const b = parseInt(expanded.slice(4, 6), 16);
   100→    return `rgba(${r}, ${g}, ${b}, ${alpha})`;
   101→  }
   102→  if (/^[0-9a-fA-F]{6}$/.test(hex)) {
   103→    const r = parseInt(hex.slice(0, 2), 16);
   104→    const g = parseInt(hex.slice(2, 4), 16);
   105→    const b = parseInt(hex.slice(4, 6), 16);
   106→    return `rgba(${r}, ${g}, ${b}, ${alpha})`;
   107→  }
   108→  return color;
   109→}

   119→  const [mode, setMode] = useState<Mode>('week');
   120→  const [cursor, setCursor] = useState<Date>(() => {
   121→    const d = new Date();
   122→    d.setHours(0, 0, 0, 0);
   123→    return d;

   125→  const [facilityFilter, setFacilityFilter] = useState<string>('all');
   126→  const [search, setSearch] = useState('');
   127→
   128→  const locale = lang === 'bm' ? 'ms-MY' : 'en-GB';
   129→
   130→  /* ----- date range based on mode ----- */
   131→  const range = useMemo(() => {
   132→    if (mode === 'day') {
   133→      const start = new Date(cursor);
   134→      start.setHours(0, 0, 0, 0);
   135→      return { start, end: addDays(start, 1) };
   136→    }
   137→    if (mode === 'week') {
   138→      const start = startOfWeek(cursor, { weekStartsOn: 1 });
   139→      return { start, end: addDays(start, 7) };
   140→    }
   141→    // month — extend grid to full Monday-Sunday weeks
   142→    const start = startOfWeek(startOfMonth(cursor), { weekStartsOn: 1 });
   143→    const end = addDays(endOfWeek(endOfMonth(cursor), { weekStartsOn: 1 }), 1);
   144→    return { start, end };
   145→  }, [mode, cursor]);
   146→
   147→  const from = range.start.toISOString();

   149→  const scope = isAuthenticated && isAdmin ? 'all' : 'public';
   150→
   151→  /* ----- queries ----- */
   152→  const { data: facilitiesData } = useQuery({
   153→    queryKey: ['facilities'],
   154→    queryFn: async () => {
   155→      const r = await fetch('/api/facilities');
   156→      const j = await r.json();
   157→      return (j.data ?? []) as Array<{
   158→        id: string;
   159→        name: string;
   160→        category: string;
   161→        colorCode: string;
   162→        location: string;
   163→        capacity: number;
   164→      }>;
   165→    },
   166→  });
   167→
   168→  const { data: bookings, isLoading } = useQuery({
   169→    queryKey: ['calendar-bookings', scope, from, to],
   170→    queryFn: async () => {
   171→      const url =
   172→        `/api/bookings?scope=${scope}` +
   173→        `&from=${encodeURIComponent(from)}` +
   174→        `&to=${encodeURIComponent(to)}` +
   175→        `&pageSize=300`;
   176→      const r = await fetch(url);
   177→      const j = await r.json();
   178→      return (j.data ?? []) as CalendarBooking[];
   179→    },
   180→  });
   181→
   182→  /* ----- group by day + detect overlap ----- */
   183→  const bookingsByDay = useMemo(() => {
   184→    const map = new Map<string, CalendarBooking[]>();
   185→    for (const b of bookings ?? []) {
   186→      const key = new Date(b.eventDate).toDateString();
   187→      if (!map.has(key)) map.set(key, []);
   188→      map.get(key)!.push(b);
   189→    }
   190→    for (const list of map.values()) {
   191→      list.sort((a, b) => a.startTime.localeCompare(b.startTime));
   192→    }
   193→    return map;
   194→  }, [bookings]);
   195→
   196→  // Visual-overlap detection: same facility, same day, time ranges intersect.
   197→  const overlapIds = useMemo(() => {
   198→    const ids = new Set<string>();
   199→    for (const list of bookingsByDay.values()) {
   200→      for (let i = 0; i < list.length; i++) {
   201→        for (let j = i + 1; j < list.length; j++) {
   202→          const a = list[i];
   203→          const b = list[j];
   204→          if (a.facility.name !== b.facility.name) continue;
   205→          if (a.startTime < b.endTime && b.startTime < a.endTime) {
   206→            ids.add(a.id);
   207→            ids.add(b.id);
   208→          }
   209→        }
   210→      }
   211→    }
   212→    return ids;
   213→  }, [bookingsByDay]);
   214→
   215→  /* ----- search + facility filter ----- */
   216→  const filteredByDay = useMemo(() => {
   217→    const map = new Map<string, CalendarBooking[]>();
   218→    const s = search.trim().toLowerCase();
   219→    for (const [k, list] of bookingsByDay.entries()) {
   220→      const filtered = list.filter((b) => {
   221→        if (facilityFilter !== 'all' && b.facility.name !== facilityFilter) return false;
   222→        if (!s) return true;
   223→        const applicant = b.user?.name?.toLowerCase() ?? '';
   224→        return (
   225→          (b.programName?.toLowerCase().includes(s) ?? false) ||
   226→          b.facility.name.toLowerCase().includes(s) ||
   227→          b.bookingRef.toLowerCase().includes(s) ||
   228→          applicant.includes(s)
   229→        );
   230→      });
   231→      map.set(k, filtered);
   232→    }
   233→    return map;
   234→  }, [bookingsByDay, facilityFilter, search]);
   235→
   236→  /* ----- navigation ----- */
   237→  const goPrev = () => {
   238→    if (mode === 'day') setCursor((c) => addDays(c, -1));
   239→    else if (mode === 'week') setCursor((c) => addDays(c, -7));
   240→    else setCursor((c) => addMonths(c, -1));
   241→  };
   242→  const goNext = () => {
   243→    if (mode === 'day') setCursor((c) => addDays(c, 1));
   244→    else if (mode === 'week') setCursor((c) => addDays(c, 7));
   245→    else setCursor((c) => addMonths(c, 1));
   246→  };
   247→  const goToday = () => {
   248→    const d = new Date();
   249→    d.setHours(0, 0, 0, 0);
   250→    setCursor(d);
   251→  };
   252→
   253→  const periodLabel = useMemo(() => {
   254→    if (mode === 'day') {
   255→      return cursor.toLocaleDateString(locale, {
   256→        weekday: 'long',
   257→        day: 'numeric',
   258→        month: 'long',
   259→        year: 'numeric',
   260→      });
   261→    }
   262→    if (mode === 'week') {
   263→      const s = startOfWeek(cursor, { weekStartsOn: 1 });
   264→      const e = endOfWeek(cursor, { weekStartsOn: 1 });
   265→      return `${s.toLocaleDateString(locale, { day: 'numeric', month: 'short' })} — ${e.toLocaleDateString(locale, {
   266→        day: 'numeric',
   267→        month: 'short',
   268→        year: 'numeric',
   269→      })}`;
   270→    }
   271→    return cursor.toLocaleDateString(locale, { month: 'long', year: 'numeric' });
   272→  }, [mode, cursor, locale]);

   276→    const monday = startOfWeek(new Date(), { weekStartsOn: 1 });
   277→    return Array.from({ length: 7 }, (_, i) => {
   278→      const d = addDays(monday, i);
   279→      return d.toLocaleDateString(locale, { weekday: 'short' });
   280→    });
   281→  }, [locale]);
   282→
   283→  /* ----- actions ----- */

   289→      lang === 'bm'
   290→        ? "Klik 'Mohon Tempahan' untuk membuat tempahan baharu"
   291→        : "Click 'New Booking' to create a new booking",
   292→    );
   293→  };
   294→
   295→  /* ----- render ----- */

   297→    <motion.div
   298→      initial={{ opacity: 0, y: 10 }}
   299→      animate={{ opacity: 1, y: 0 }}
   300→      transition={{ duration: 0.3 }}
   301→      className="space-y-4"

   303→      {/* Hero */}

   305→        <div className="absolute top-0 right-0 w-64 h-64 rounded-full bg-teal-500/10 blur-3xl pointer-events-none" />
   306→        <div className="absolute -bottom-12 -left-12 w-48 h-48 rounded-full bg-amber-500/10 blur-3xl pointer-events-none" />
   307→        <div className="relative z-10">
   308→          <div className="flex items-center gap-2 text-[11px] text-muted-foreground uppercase tracking-wider mb-2">
   309→            <Calendar className="w-3 h-3" />
   310→            <span>{lang === 'bm' ? 'Paparan Bersepadu' : 'Integrated View'} · FR-05</span>
   311→          </div>
   312→          <h2 className="text-xl lg:text-2xl font-bold gradient-text mb-1">{t('nav_calendar')}</h2>
   313→          <p className="text-sm text-muted-foreground max-w-2xl">
   314→            {lang === 'bm'
   315→              ? 'Paparan kalendar bersepadu semua tempahan kemudahan PLTT — mod Hari / Minggu / Bulan dengan kod warna per kemudahan.'

   319→          <div className="flex flex-wrap gap-2 mt-4">

   321→              <span
   322→                key={f.id}
   323→                className="inline-flex items-center gap-1.5 text-[10px] px-2 py-1 rounded-full glass-input border border-border/40"

   325→                <span className="w-2 h-2 rounded-full" style={{ background: f.colorCode }} />
   326→                {f.name}
   327→              </span>
   328→            ))}
   329→            {scope === 'all' && (
   330→              <Badge variant="outline" className="text-[10px] glass-input border-amber-400/40 text-amber-700">
   331→                {lang === 'bm' ? 'Skop Pentadbir' : 'Admin Scope'}
   332→              </Badge>
   333→            )}
   334→          </div>
   335→        </div>
   336→      </GlassCard>
   337→
   338→      {/* Toolbar */}
   339→      <GlassCard className="p-3 lg:p-4">
   340→        <div className="flex flex-col gap-3">
   341→          <div className="flex flex-col lg:flex-row lg:items-center gap-3">
   342→            {/* Mode switcher */}
   343→            <div
   344→              role="tablist"
   345→              aria-label="Calendar mode"
   346→              className="inline-flex p-1 rounded-lg glass-input border border-border/40 self-start"
   347→            >

   360→                <ChevronRight className="w-4 h-4" />
   361→              </Button>
   362→              <Button variant="outline" size="sm" onClick={goToday} className="ml-1 glass-input text-xs shrink-0">
   363→                {t('calendar_today')}
   364→              </Button>
   365→            </div>
   366→
   367→            {/* Export buttons */}
   368→            <div className="flex items-center gap-2 shrink-0">
   369→              <Button variant="outline" size="sm" onClick={() => handleExport('pdf')} className="glass-input text-xs">
   370→                <FileText className="w-3.5 h-3.5" />
   371→                PDF
   372→              </Button>
   373→              <Button variant="outline" size="sm" onClick={() => handleExport('excel')} className="glass-input text-xs">
   374→                <FileSpreadsheet className="w-3.5 h-3.5" />
   375→                Excel
   376→              </Button>
   377→            </div>

   379→
   380→          {/* Search + facility filter */}
   381→          <div className="flex flex-col md:flex-row gap-3">
   382→            <div className="relative flex-1">
   383→              <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground pointer-events-none" />
   384→              <Input
   385→                value={search}
   386→                onChange={(e) => setSearch(e.target.value)}
   387→                placeholder={
   388→                  lang === 'bm'
   389→                    ? 'Cari nama program, kemudahan, rujukan…'
   390→                    : 'Search program, facility, reference…'
   391→                }
   392→                className="pl-9 glass-input"
   393→                aria-label={t('search')}
   394→              />
   395→            </div>
   396→            <Select value={facilityFilter} onValueChange={setFacilityFilter}>
   397→              <SelectTrigger className="glass-input md:w-64" aria-label={t('filter')}>
   398→                <SelectValue placeholder={t('form_facility')} />
   399→              </SelectTrigger>
   400→              <SelectContent className="glass-strong">
   401→                <SelectItem value="all">{lang === 'bm' ? 'Semua Kemudahan' : 'All Facilities'}</SelectItem>
   402→                {(facilitiesData ?? []).map((f) => (
   403→                  <SelectItem key={f.id} value={f.name}>
   404→                    <span className="inline-flex items-center gap-2">
   405→                      <span className="w-2 h-2 rounded-full" style={{ background: f.colorCode }} />
   406→                      {f.name}
   407→                    </span>
   408→                  </SelectItem>
   409→                ))}
   410→              </SelectContent>
   411→            </Select>
   412→          </div>
   413→        </div>
   414→      </GlassCard>
   415→
   416→      {/* Calendar body */}
   417→      {isLoading ? (
   418→        <CalendarSkeleton mode={mode} />
   419→      ) : mode === 'month' ? (
   420→        <MonthGrid
   421→          cursor={cursor}
   422→          locale={locale}
   423→          weekdayLabels={weekdayLabels}
   424→          filteredByDay={filteredByDay}
   425→          overlapIds={overlapIds}
   426→          onEmptyClick={handleEmptyClick}
   427→          scope={scope}
   428→          lang={lang}
   429→        />
   430→      ) : mode === 'week' ? (
   431→        <WeekGrid
   432→          cursor={cursor}
   433→          locale={locale}
   434→          filteredByDay={filteredByDay}
   435→          overlapIds={overlapIds}
   436→          onEmptyClick={handleEmptyClick}
   437→          scope={scope}
   438→          lang={lang}
   439→        />
   440→      ) : (
   441→        <DayTimeline
   442→          cursor={cursor}
   443→          locale={locale}
   444→          filteredByDay={filteredByDay}
   445→          overlapIds={overlapIds}
   446→          onEmptyClick={handleEmptyClick}
   447→          scope={scope}
   448→          lang={lang}
   449→        />
   450→      )}
   451→    </motion.div>
   452→  );
   453→}

   489→/* ---------- Month ---------- */
   490→
   491→function MonthGrid({
   492→  cursor,
   493→  locale,
   494→  weekdayLabels,
   495→  filteredByDay,
   496→  overlapIds,
   497→  onEmptyClick,
   498→  scope,
   499→  lang,
   500→}: {
   501→  cursor: Date;
   502→  locale: string;
   503→  weekdayLabels: string[];
   504→  filteredByDay: Map<string, CalendarBooking[]>;
   505→  overlapIds: Set<string>;
   506→  onEmptyClick: () => void;
   507→  scope: string;
   508→  lang: 'bm' | 'en';
   509→}) {
   510→  const monthStart = startOfMonth(cursor);
   511→  const monthEnd = endOfMonth(cursor);
   512→  const gridStart = startOfWeek(monthStart, { weekStartsOn: 1 });
   513→  const gridEnd = endOfWeek(monthEnd, { weekStartsOn: 1 });
   514→  const days = eachDayOfInterval({ start: gridStart, end: gridEnd });
   515→
   516→  return (
   517→    <GlassCard className="p-3 lg:p-4">
   518→      {/* Weekday header */}
   519→      <div className="grid grid-cols-7 gap-1 mb-2">
   520→        {weekdayLabels.map((d) => (
   521→          <div
   522→            key={d}
   523→            className="text-[11px] font-semibold uppercase tracking-wide text-muted-foreground text-center py-1"
   524→          >
   525→            {d}
   526→          </div>
   527→        ))}
   528→      </div>
   529→      {/* Day cells */}
   530→      <div className="grid grid-cols-7 gap-1">
   531→        {days.map((day) => {
   532→          const inMonth = isSameMonth(day, cursor);
   533→          const today = isTodayFn(day);
   534→          const list = filteredByDay.get(day.toDateString()) ?? [];
   535→          const visible = list.slice(0, 3);
   536→          const overflow = list.length - visible.length;
   537→          const isEmpty = list.length === 0;
   538→          return (
   539→            <div
   540→              key={day.toISOString()}
   541→              role={isEmpty ? 'button' : undefined}
   542→              tabIndex={isEmpty ? 0 : undefined}
   543→              onClick={() => isEmpty && onEmptyClick()}
   544→              onKeyDown={(e) => {

   843→        {GUTTER_HOURS.slice(0, -1).map((h) => {
   844→          const top = (h - DAY_START_HOUR) * 60 * CELL_HEIGHT_PER_MIN;

   846→            <div
   847→              key={h}
