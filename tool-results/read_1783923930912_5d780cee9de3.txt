   107→   107→// ===================== Helpers =====================
   108→   108→const safeParseEquip = (raw?: string | null): string[] => {
   109→   109→  if (!raw) return [];
   110→   110→  // Try JSON array first
   111→   111→  try {
   112→   112→    const v = JSON.parse(raw);
   113→   113→    if (Array.isArray(v)) return v.filter((x) => typeof x === 'string' && x.trim());
   114→   114→  } catch {
   115→   115→    // fallthrough — treat as comma-separated
   116→   116→  }
   117→   117→  if (typeof raw === 'string' && raw.trim()) {
   118→   118→    return raw
   119→   119→      .split(/[,;]+/)
   120→   120→      .map((s) => s.trim())
   121→   121→      .filter(Boolean);
   122→   122→  }
   123→   123→  return [];
   124→   124→};

   887→   887→                {tr(

   907→   907→      </AlertDialog>
   908→   908→
   909→   909→      {/* Footer info */}
   910→   910→      {user && (
   911→   911→        <p className="text-xs text-muted-foreground text-center pt-2">
   912→   912→          {tr('Log masuk sebagai', 'Logged in as')}{' '}
   913→   913→          <span className="font-medium text-foreground/80">{user.name}</span>{' '}
   914→   914→          ·{' '}
   915→   915→          {user.role === 'SUPER_ADMIN'
   916→   916→            ? t('role_SUPER_ADMIN')
   917→   917→            : user.role === 'FACILITY_ADMIN'
   918→   918→              ? t('role_FACILITY_ADMIN')
   919→   919→              : t('role_USER')}
   920→   920→        </p>
   921→   921→      )}
   922→   922→    </motion.div>
   923→   923→  );
   924→   924→}
