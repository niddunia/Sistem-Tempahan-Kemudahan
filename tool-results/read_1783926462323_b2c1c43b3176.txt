    69→
    70→interface Facility {
    71→  id: string;
    72→  name: string;
    73→  category: Category;
    74→  capacity: number;
    75→  location?: string | null;
    76→  equipment?: string | null;

    78→  operatingEnd?: string | null;
    79→  colorCode: string;
    80→  status: FacilityStatus;
    81→  createdAt: string;
    82→  updatedAt: string;

    85→interface FacilityFormState {
    86→  name: string;
    87→  category: Category;
    88→  capacity: string;
    89→  location: string;
    90→  equipment: string;
    91→  operatingStart: string;
    92→  operatingEnd: string;
    93→  colorCode: string;
    94→}
    95→
    96→const DEFAULT_FORM: FacilityFormState = {
    97→  name: '',
    98→  category: 'COMPUTER_ROOM',
    99→  capacity: '30',
   100→  location: '',
   101→  equipment: '',
   102→  operatingStart: '08:00',
   103→  operatingEnd: '22:00',
   104→  colorCode: '#0d9488',
   105→};
   106→
   107→// ===================== Helpers =====================
   108→const safeParseEquip = (raw?: string | null): string[] => {
   109→  if (!raw) return [];
   110→  // Try JSON array first
   111→  try {
   112→    const v = JSON.parse(raw);
   113→    if (Array.isArray(v)) return v.filter((x) => typeof x === 'string' && x.trim());
   114→  } catch {
   115→    // fallthrough — treat as comma-separated
   116→  }
   117→  if (typeof raw === 'string' && raw.trim()) {
   118→    return raw
   119→      .split(/[,;]+/)
   120→      .map((s) => s.trim())
   121→      .filter(Boolean);
   122→  }
   123→  return [];
   124→};
   125→
   126→const toFormState = (f: Facility): FacilityFormState => ({
   127→  name: f.name ?? '',
   128→  category: f.category ?? 'COMPUTER_ROOM',
   129→  capacity: String(f.capacity ?? 0),
   130→  location: f.location ?? '',
   131→  equipment: safeParseEquip(f.equipment).join(', '),
   132→  operatingStart: f.operatingStart ?? '08:00',
   133→  operatingEnd: f.operatingEnd ?? '22:00',
   134→  colorCode: f.colorCode ?? '#0d9488',
   135→});

   205→      return true;

   211→    mutationFn: async (data: FacilityFormState) => {
   212→      const body = {
   213→        name: data.name.trim(),
   214→        category: data.category,
   215→        capacity: parseInt(data.capacity, 10) || 0,
   216→        location: data.location.trim() || undefined,
   217→        equipment: data.equipment.trim() || undefined,
   218→        operatingStart: data.operatingStart,

   224→        headers: { 'Content-Type': 'application/json' },

   243→        capacity: parseInt(data.capacity, 10) || 0,
   244→        location: data.location.trim() || undefined,

   246→        operatingStart: data.operatingStart,
   247→        operatingEnd: data.operatingEnd,
   248→        colorCode: data.colorCode,
   249→      };

   251→        method: 'PATCH',
   252→        headers: { 'Content-Type': 'application/json' },
   253→        body: JSON.stringify(body),
   254→      });

   330→      updateMutation.mutate({ id: editTarget.id, data: form });

   332→      createMutation.mutate(form);
   333→    }
   334→  };
   335→
   336→  const cycleStatus = (f: Facility) => {
   337→    // ACTIVE → MAINTENANCE → INACTIVE → ACTIVE
   338→    const next: FacilityStatus =
   339→      f.status === 'ACTIVE' ? 'MAINTENANCE' : f.status === 'MAINTENANCE' ? 'INACTIVE' : 'ACTIVE';
   340→    if (next === 'INACTIVE') {
   341→      // require confirmation
   342→      setStatusCycleTarget(f);
   343→    } else {
   344→      statusMutation.mutate({ id: f.id, status: next });

   346→  };

   360→        <div className="relative flex flex-col sm:flex-row sm:items-center sm:justify-between gap-4">

   363→              <Building2 className="w-6 h-6" />

   366→              <h1 className="text-2xl lg:text-3xl font-bold gradient-text leading-tight">
   367→                {t('nav_facilities')}

   369→              <p className="text-sm text-muted-foreground mt-1">
   370→                {tr('Tambah, kemas kini dan urus kemudahan bilik komputer & dewan kuliah', 'Add, update and manage computer room & lecture hall facilities')}
   371→              </p>
   372→            </div>
   373→          </div>

   375→            <Button variant="outline" size="sm" onClick={() => setView('dashboard')} className="glass-input">
   376→              {tr('Kembali', 'Back')}
   377→            </Button>
   378→            {isSuperAdmin && (
   379→              <Button
   380→                onClick={openCreate}
   381→                className="gradient-primary text-white shadow-md hover:shadow-lg transition-shadow"
   382→                size="sm"
   383→              >
   384→                <PlusCircle className="w-4 h-4 mr-2" />
   385→                {tr('Tambah Kemudahan', 'Add Facility')}
   386→              </Button>
   387→            )}
   388→          </div>
   389→        </div>
   390→      </GlassCard>
   391→
   392→      {/* Access warning for non-super-admin */}
   393→      {!isSuperAdmin && (
   394→        <GlassCard className="p-4 border-amber-300/60 bg-amber-50/60">
   395→          <div className="flex items-start gap-3 text-amber-800">
   396→            <ShieldCheck className="w-5 h-5 mt-0.5 shrink-0" />
   397→            <div className="text-sm">
   398→              <p className="font-semibold">{tr('Akses Terhad', 'Restricted Access')}</p>
   399→              <p className="text-amber-700/90">
   400→                {tr('Hanya Pentadbir Sistem boleh mengubah kemudahan. Anda hanya boleh melihat.', 'Only System Admins can modify facilities. You can view only.')}

   408→      <div className="grid grid-cols-2 lg:grid-cols-5 gap-3">
   409→        <StatTile
   410→          icon={<Building2 className="w-5 h-5" />}
   411→          label={tr('Jumlah Kemudahan', 'Total Facilities')}
   412→          value={stats.total}
   413→          tone="teal"
   414→          loading={isLoading}

   419→          value={stats.active}

   421→          loading={isLoading}

   502→
   503→        {isLoading ? (

   566→                        <span className="inline-flex items-center gap-1 text-sm font-medium">

   574→                            <MapPin className="w-3.5 h-3.5" />

   636→                            variant="outline"
   637→                            size="sm"

   639→                            disabled={!isSuperAdmin}
   640→                            className="h-8 w-8 p-0 text-rose-600 hover:bg-rose-50 hover:text-rose-700 border-rose-200/60"

   642→                            aria-label={t('delete')}
   643→                          >
   644→                            <Trash2 className="w-3.5 h-3.5" />
   645→                          </Button>
   646→                        </div>
   647→                      </TableCell>
   648→                    </TableRow>
   649→                  );
   650→                })}
   651→              </TableBody>
   652→            </Table>
   653→          </div>
   654→        )}
   655→      </GlassCard>
   656→
   657→      {/* Create/Edit Dialog */}
   658→      <Dialog open={dialogOpen} onOpenChange={(o) => setDialogOpen(o)}>
   659→        <DialogContent className="sm:max-w-2xl max-h-[90vh] overflow-y-auto scroll-area-thin">
   660→          <DialogHeader>
   661→            <DialogTitle className="flex items-center gap-2">
   662→              <Building2 className="w-5 h-5 text-teal-600" />
   663→              {editTarget
   664→                ? tr('Kemas Kini Kemudahan', 'Edit Facility')
   665→                : tr('Tambah Kemudahan Baharu', 'Add New Facility')}
   666→            </DialogTitle>
   667→            <DialogDescription>
   668→              {editTarget
   669→                ? tr('Kemas kini maklumat kemudahan ini.', 'Update this facility information.')
   670→                : tr('Isi butiran kemudahan baru.', 'Fill in the new facility details.')}
   671→            </DialogDescription>
   672→          </DialogHeader>
   673→
   674→          <div className="grid gap-4 py-2">
   675→            <div className="grid gap-2">
   676→              <Label htmlFor="f-name">{t('name')} *</Label>
   677→              <Input
   678→                id="f-name"
   679→                value={form.name}
   680→                onChange={(e) => setForm({ ...form, name: e.target.value })}
   681→                placeholder={tr('cth. Bilik Komputer 1', 'e.g. Computer Room 1')}
   682→                className="glass-input"
   683→              />
   684→            </div>
   685→
   686→            <div className="grid grid-cols-2 gap-4">
   687→              <div className="grid gap-2">
   688→                <Label>{tr('Kategori', 'Category')} *</Label>
   689→                <Select
   690→                  value={form.category}
   691→                  onValueChange={(v) => setForm({ ...form, category: v as Category })}
   692→                  disabled={!!editTarget}
   693→                >
   694→                  <SelectTrigger className="w-full glass-input">
   695→                    <SelectValue />
   696→                  </SelectTrigger>
   697→                  <SelectContent>
   698→                    <SelectItem value="COMPUTER_ROOM">{tr('Bilik Komputer', 'Computer Room')}</SelectItem>
   699→                    <SelectItem value="LECTURE_HALL">{tr('Dewan Kuliah', 'Lecture Hall')}</SelectItem>
   700→                  </SelectContent>
   701→                </Select>
   702→                {editTarget && (
   703→                  <p className="text-xs text-muted-foreground">
   704→                    {tr('Kategori tidak boleh diubah selepas dibuat.', 'Category cannot be changed after creation.')}
   705→                  </p>
   706→                )}
   707→              </div>
   708→              <div className="grid gap-2">
   709→                <Label htmlFor="f-cap">{tr('Kapasiti', 'Capacity')} *</Label>
   710→                <Input
   711→                  id="f-cap"
   712→                  type="number"
   713→                  min={1}
   714→                  max={1000}
   715→                  value={form.capacity}
   716→                  onChange={(e) => setForm({ ...form, capacity: e.target.value })}
   717→                  className="glass-input"
   718→                />
   719→              </div>
   720→            </div>
   721→
   722→            <div className="grid gap-2">
   723→              <Label htmlFor="f-loc">{tr('Lokasi', 'Location')}</Label>
   724→              <Input
   725→                id="f-loc"
   726→                value={form.location}
   727→                onChange={(e) => setForm({ ...form, location: e.target.value })}
   728→                placeholder={tr('cth. Aras 2, Blok A', 'e.g. Level 2, Block A')}
   729→                className="glass-input"
   730→              />
   731→            </div>
   732→
   733→            <div className="grid gap-2">
   734→              <Label htmlFor="f-equip">{tr('Peralatan', 'Equipment')}</Label>
   735→              <Textarea
   736→                id="f-equip"
   737→                value={form.equipment}
   738→                onChange={(e) => setForm({ ...form, equipment: e.target.value })}
   739→                placeholder={tr('Pisahkan dengan koma: LCD Projector, Whiteboard, AC, WiFi', 'Separate with commas: LCD Projector, Whiteboard, AC, WiFi')}

   741→                className="glass-input resize-none"
   742→              />
   743→              {form.equipment.trim() && (
   744→                <div className="flex flex-wrap gap-1 mt-1">
   745→                  {form.equipment
   746→                    .split(/[,;]+/)
   747→                    .map((s) => s.trim())
   748→                    .filter(Boolean)
   749→                    .map((e, i) => (
   750→                      <span
   751→                        key={i}
   752→                        className="inline-flex items-center text-[10px] px-1.5 py-0.5 rounded-full bg-emerald-50/80 text-emerald-700 border border-emerald-200/60"
   753→                      >
   754→                        {e}
   755→                      </span>
   756→                    ))}
   757→                </div>
   758→              )}
   759→            </div>
   760→
   761→            <div className="grid grid-cols-2 gap-4">
   762→              <div className="grid gap-2">
   763→                <Label htmlFor="f-start">{tr('Waktu Mula', 'Start Time')}</Label>
   764→                <Input
   765→                  id="f-start"
   766→                  type="time"
   767→                  value={form.operatingStart}
   768→                  onChange={(e) => setForm({ ...form, operatingStart: e.target.value })}
   769→                  className="glass-input"
   770→                />
   771→              </div>
   772→              <div className="grid gap-2">
   773→                <Label htmlFor="f-end">{tr('Waktu Tamat', 'End Time')}</Label>
   774→                <Input
   775→                  id="f-end"
   776→                  type="time"
   777→                  value={form.operatingEnd}
   778→                  onChange={(e) => setForm({ ...form, operatingEnd: e.target.value })}
   779→                  className="glass-input"
   780→                />
   781→              </div>
   782→            </div>
   783→
   784→            <div className="grid gap-2">
   785→              <Label htmlFor="f-color">{tr('Warna Penanda', 'Color Marker')}</Label>
   786→              <div className="flex items-center gap-3">
   787→                <input
   788→                  id="f-color"
   789→                  type="color"
   790→                  value={form.colorCode}
   791→                  onChange={(e) => setForm({ ...form, colorCode: e.target.value })}
   792→                  className="w-12 h-10 rounded-md border border-border bg-transparent cursor-pointer p-1"
   793→                  aria-label={tr('Pemilih warna', 'Color picker')}
   794→                />
   795→                <Input
   796→                  value={form.colorCode}
   797→                  onChange={(e) => setForm({ ...form, colorCode: e.target.value })}
   798→                  className="glass-input w-32 font-mono text-sm"
   799→                />
   800→                <div
   801→                  className="ml-2 inline-flex items-center gap-2 text-xs text-muted-foreground"
   802→                >
   803→                  <span
   804→                    className="inline-block w-4 h-4 rounded-full ring-2 ring-white shadow"
   805→                    style={{ backgroundColor: form.colorCode }}
   806→                    aria-hidden
   807→                  />
   808→                  {tr('Pratonton', 'Preview')}
   809→                </div>
   810→              </div>
   811→            </div>
   812→          </div>
   813→
   814→          <DialogFooter className="gap-2">
   815→            <Button variant="outline" onClick={() => setDialogOpen(false)} className="glass-input">
   816→              {t('close')}
   817→            </Button>
   818→            <Button
   819→              onClick={submitForm}
   820→              disabled={createMutation.isPending || updateMutation.isPending}
   821→              className="gradient-primary text-white"
   822→            >
   823→              {(createMutation.isPending || updateMutation.isPending) && (

   825→              )}
   826→              {editTarget ? t('save') : tr('Tambah', 'Add')}
   827→            </Button>
   828→          </DialogFooter>
   829→        </DialogContent>
   830→      </Dialog>
   831→

   835→        onOpenChange={(o) => !o && setStatusCycleTarget(null)}
   836→      >

   838→          <AlertDialogHeader>
   839→            <AlertDialogTitle className="flex items-center gap-2">
   840→              <Wrench className="w-5 h-5 text-amber-600" />
   841→              {tr('Nyahaktifkan Kemudahan', 'Deactivate Facility')}
   842→            </AlertDialogTitle>
   843→            <AlertDialogDescription>

   849→                {tr('Kemudahan tidak akan tersedia untuk tempahan baharu.', 'The facility will be unavailable for new bookings.')}

   851→            </AlertDialogDescription>
   852→          </AlertDialogHeader>
   853→          <AlertDialogFooter>
   854→            <AlertDialogCancel>{t('no')}</AlertDialogCancel>
   855→            <AlertDialogAction

   858→                  statusMutation.mutate({ id: statusCycleTarget.id, status: 'INACTIVE' });
   859→                }
   860→              }}
   861→              disabled={statusMutation.isPending}
   862→              className="bg-amber-600 hover:bg-amber-700 text-white"
   863→            >
   864→              {statusMutation.isPending ? t('loading') : t('yes')}

   866→          </AlertDialogFooter>
   867→        </AlertDialogContent>
   868→      </AlertDialog>
   869→
   870→      {/* Delete Confirmation */}
   871→      <AlertDialog
   872→        open={!!deleteTarget}
   873→        onOpenChange={(o) => !o && setDeleteTarget(null)}
   874→      >
   875→        <AlertDialogContent>
   876→          <AlertDialogHeader>
   877→            <AlertDialogTitle className="flex items-center gap-2 text-rose-700">
   878→              <Trash2 className="w-5 h-5" />
   879→              {tr('Padam Kemudahan', 'Delete Facility')}
   880→            </AlertDialogTitle>
   881→            <AlertDialogDescription>
   882→              {tr(
   883→                `Adakah anda pasti mahu memadam "${deleteTarget?.name ?? ''}"?`,
   884→                `Are you sure you want to delete "${deleteTarget?.name ?? ''}"?`,
   885→              )}
   886→              <span className="block mt-2 text-rose-700">
   887→                {tr(
   888→                  'Tindakan ini akan menetapkan status kemudahan kepada Tidak Aktif (padam lembut).',
   889→                  'This action sets the facility status to Inactive (soft delete).',
   890→                )}
   891→              </span>
   892→            </AlertDialogDescription>
   893→          </AlertDialogHeader>
   894→          <AlertDialogFooter>
   895→            <AlertDialogCancel>{t('no')}</AlertDialogCancel>
   896→            <AlertDialogAction
   897→              onClick={() => {
   898→                if (deleteTarget) deleteMutation.mutate(deleteTarget.id);
   899→              }}
   900→              disabled={deleteMutation.isPending}
   901→              className="bg-rose-600 hover:bg-rose-700 text-white"
   902→            >
   903→              {deleteMutation.isPending ? t('loading') : t('delete')}
   904→            </AlertDialogAction>
   905→          </AlertDialogFooter>
   906→        </AlertDialogContent>
   907→      </AlertDialog>
   908→
   909→      {/* Footer info */}
   910→      {user && (
   911→        <p className="text-xs text-muted-foreground text-center pt-2">
   912→          {tr('Log masuk sebagai', 'Logged in as')}{' '}
   913→          <span className="font-medium text-foreground/80">{user.name}</span>{' '}
   914→          ·{' '}
   915→          {user.role === 'SUPER_ADMIN'
   916→            ? t('role_SUPER_ADMIN')
   917→            : user.role === 'FACILITY_ADMIN'
   918→              ? t('role_FACILITY_ADMIN')
   919→              : t('role_USER')}
   920→        </p>
   921→      )}
   922→    </motion.div>
   923→  );
   924→}

   926→// ===================== Stat Tile =====================
   927→function StatTile({
   928→  icon,
   929→  label,
   930→  value,
   931→  tone,
   932→  loading,
   933→}: {
   934→  icon: React.ReactNode;
   935→  label: string;
   936→  value: number;
   937→  tone: 'teal' | 'emerald' | 'amber' | 'zinc';
   938→  loading?: boolean;
   939→}) {
   940→  const toneClasses: Record<string, string> = {
   941→    teal: 'bg-teal-500/15 text-teal-700 dark:text-teal-300',
   942→    emerald: 'bg-emerald-500/15 text-emerald-700 dark:text-emerald-300',
   943→    amber: 'bg-amber-500/15 text-amber-700 dark:text-amber-300',
   944→    zinc: 'bg-zinc-500/15 text-zinc-700 dark:text-zinc-300',
   945→  };
   946→  return (
   947→    <GlassCard className="p-4 flex items-center gap-3">
   948→      <div className={cn('grid place-items-center w-10 h-10 rounded-xl shrink-0', toneClasses[tone])}>
   949→        {icon}
   950→      </div>
   951→      <div className="min-w-0">
   952→        <p className="text-[11px] text-muted-foreground truncate uppercase tracking-wide">{label}</p>
   953→        {loading ? (
   954→          <Skeleton className="h-6 w-12 mt-1" />
   955→        ) : (
   956→          <p className="text-xl font-bold leading-tight">{value.toLocaleString()}</p>
   957→        )}
   958→      </div>
   959→    </GlassCard>
   960→  );
   961→}
