    96→// ===================== Helpers =====================
    97→const safeParseEquip = (raw?: string | null): string[] => {
    98→  if (!raw) return [];
    99→  try {
   100→    const v = JSON.parse(raw);
   101→    if (Array.isArray(v)) return v.filter((x) => typeof x === 'string');
   102→    if (typeof v === 'string') return [v];
   103→  } catch {
   104→    return raw.split(',').map((s) => s.trim()).filter(Boolean);
   105→  }
   106→  return [];
   107→};
   108→
   109→const fmtDate = (iso: string, lang: 'bm' | 'en') =>
   110→  new Date(iso).toLocaleDateString(lang === 'bm' ? 'ms-MY' : 'en-GB', {
   111→    weekday: 'short',
   112→    day: 'numeric',
   113→    month: 'short',
   114→    year: 'numeric',
   115→  });
   116→
   117→const fmtDateTime = (iso: string, lang: 'bm' | 'en') =>
   118→  new Date(iso).toLocaleString(lang === 'bm' ? 'ms-MY' : 'en-GB', {
   119→    day: 'numeric',
   120→    month: 'short',
   121→    year: 'numeric',
   122→    hour: '2-digit',
   123→    minute: '2-digit',
   124→  });
   125→
   126→const isSameDay = (iso: string, ref: Date) => {
   127→  const d = new Date(iso);
   128→  return (
   129→    d.getFullYear() === ref.getFullYear() &&
   130→    d.getMonth() === ref.getMonth() &&
   131→    d.getDate() === ref.getDate()
   132→  );
   133→};

   135→// ===================== Main View =====================
   136→export function ApprovalsView() {
   137→  const { t, lang } = useT();
   138→  useApp(); // ensure store ready / language reactive
   139→  const { user } = useCurrentUser();
   140→  const qc = useQueryClient();
   141→
   142→  const [tab, setTab] = useState<'PENDING' | 'NEEDS_INFO'>('PENDING');
   143→  const [search, setSearch] = useState('');
   144→  const [selectedId, setSelectedId] = useState<string | null>(null);
   145→  const [dialogAction, setDialogAction] = useState<ActionKind | null>(null);
   146→  const [comment, setComment] = useState('');
   147→
   148→  // ---- Fetch pending / needs-info approvals ----
   149→  const { data: bookings, isLoading } = useQuery<Booking[]>({
   150→    queryKey: ['approvals', tab],
   151→    queryFn: async () => {
   152→      const r = await fetch(`/api/approvals?status=${tab}`, { credentials: 'include' });
   153→      if (!r.ok) throw new Error('failed');
   154→      const j = await r.json();
   155→      return j.data as Booking[];
   156→    },
   157→    enabled: !!user,
   158→  });
   159→
   160→  // ---- Filter list (search by bookingRef / applicant / facility) ----
   161→  const filtered = useMemo(() => {
   162→    if (!bookings) return [];
   163→    const s = search.trim().toLowerCase();
   164→    if (!s) return bookings;
   165→    return bookings.filter((b) =>
   166→      b.bookingRef.toLowerCase().includes(s) ||
   167→      b.user.name.toLowerCase().includes(s) ||
   168→      b.facility.name.toLowerCase().includes(s),
   169→    );
   170→  }, [bookings, search]);
   171→
   172→  // ---- Stats ----
   173→  const stats = useMemo(() => {
   174→    const pending = bookings?.filter((b) => b.status === 'PENDING') ?? [];
   175→    const today = new Date();
   176→    const todayPending = pending.filter((b) => isSameDay(b.eventDate, today)).length;
   177→    const multiLevel = pending.filter((b) => b.requiresMultiLevel).length;
   178→    return {
   179→      totalPending: pending.length || (tab === 'PENDING' ? bookings?.length ?? 0 : 0),
   180→      todayPending,
   181→      multiLevel,
   182→    };
   183→  }, [bookings, tab]);
   184→
   185→  // ---- Selected booking object (derived; falls back to first in list) ----
   186→  const selected = useMemo(
   187→    () => filtered.find((b) => b.id === selectedId) ?? filtered[0] ?? null,
   188→    [filtered, selectedId],
   189→  );
   190→
   191→  // Track the effective selected id for downstream keys/queries
   192→  const effectiveSelectedId = selected?.id ?? null;
   193→
   194→  // ---- Conflict check for selected booking ----
   195→  const { data: conflict, isLoading: conflictLoading } = useQuery<ConflictResult>({
   196→    queryKey: ['conflict-check', effectiveSelectedId],
   197→    queryFn: async () => {
   198→      if (!selected) return { hasConflict: false, conflictingBookings: [] };
   199→      const r = await fetch('/api/bookings/check-conflict', {
   200→        method: 'POST',
   201→        headers: { 'Content-Type': 'application/json' },
   202→        credentials: 'include',
   203→        body: JSON.stringify({
   204→          facilityId: selected.facility.id,
   205→          eventDate: selected.eventDate,
   206→          startTime: selected.startTime,
   207→          endTime: selected.endTime,
   208→          excludeBookingId: selected.id,
   209→        }),
   210→      });
   211→      if (!r.ok) throw new Error('conflict-check failed');
   212→      const j = await r.json();
   213→      return j.data as ConflictResult;
   214→    },
   215→    enabled: !!selected,
   216→  });
   217→
   218→  // ---- Action mutation ----
   219→  const actionMut = useMutation({
   220→    mutationFn: async (vars: { id: string; action: ActionKind; comment: string }) => {
   221→      const r = await fetch(`/api/approvals/${vars.id}`, {
   222→        method: 'PATCH',
   223→        headers: { 'Content-Type': 'application/json' },
   224→        credentials: 'include',
   225→        body: JSON.stringify({ action: vars.action, comment: vars.comment || undefined }),
   226→      });
   227→      if (r.status === 409) {
   228→        const j = await r.json().catch(() => ({}));
   229→        throw new Error(j.message ?? (lang === 'bm' ? 'Pertindihan dikesan.' : 'Conflict detected.'));
   230→      }
   231→      if (!r.ok) {
   232→        const j = await r.json().catch(() => ({}));
   233→        throw new Error(j.message ?? (lang === 'bm' ? 'Tindakan gagal.' : 'Action failed.'));
   234→      }
   235→      return r.json();
   236→    },
   237→    onSuccess: (_data, vars) => {
   238→      const msg =
   239→        vars.action === 'APPROVE'
   240→          ? lang === 'bm' ? 'Permohonan diluluskan.' : 'Booking approved.'
   241→          : vars.action === 'REJECT'
   242→          ? lang === 'bm' ? 'Permohonan ditolak.' : 'Booking rejected.'
   243→          : lang === 'bm' ? 'Permintaan maklumat dihantar.' : 'Info request sent.';
   244→      toast.success(msg);
   245→      // refresh both lists & related caches
   246→      qc.invalidateQueries({ queryKey: ['approvals'] });
   247→      qc.invalidateQueries({ queryKey: ['conflict-check'] });
   248→      qc.invalidateQueries({ queryKey: ['notifications'] });
   249→      // auto-select next pending item (clear current selection so the
   250→      // derived `selected` falls back to the next item in the refreshed list)
   251→      setSelectedId((prev) => (prev === vars.id ? null : prev));
   252→    },
   253→    onError: (err: unknown) => {
   254→      const msg = err instanceof Error ? err.message : 'Error';
   255→      toast.error(msg);
   256→    },
   257→    onSettled: () => {
   258→      setDialogAction(null);
   259→      setComment('');
   260→    },
   261→  });
   262→
   263→  // ---- Open dialog with action ----
   264→  const openAction = (a: ActionKind) => {
   265→    setComment('');
   266→    setDialogAction(a);
   267→  };
   268→
   269→  // ---- Submit dialog ----
   270→  const submitAction = () => {
   271→    if (!selected || !dialogAction) return;
   272→    if ((dialogAction === 'REJECT' || dialogAction === 'REQUEST_INFO') && !comment.trim()) {
   273→      toast.error(lang === 'bm' ? 'Sila isi ruangan ulasan.' : 'Please fill in the comment.');
   274→      return;
   275→    }
   276→    actionMut.mutate({ id: selected.id, action: dialogAction, comment: comment.trim() });
   277→  };
   278→
   279→  // ---- Capacity check ----
   280→  const overCapacity = selected ? selected.participantCount > selected.facility.capacity : false;
   281→
   282→  return (
   283→    <motion.div
   284→      initial={{ opacity: 0, y: 10 }}
   285→      animate={{ opacity: 1, y: 0 }}
   286→      transition={{ duration: 0.3 }}
   287→      className="space-y-4"
   288→    >
   289→      {/* ===== Header / Stats ===== */}
   290→      <GlassCard className="p-5 lg:p-6 relative overflow-hidden">
   291→        <div className="absolute top-0 right-0 w-64 h-64 rounded-full bg-teal-500/10 blur-3xl pointer-events-none" />
   292→        <div className="absolute -bottom-12 -left-12 w-48 h-48 rounded-full bg-amber-500/10 blur-3xl pointer-events-none" />
   293→        <div className="relative z-10">
   294→          <div className="flex items-center gap-2 text-[11px] text-muted-foreground uppercase tracking-wider mb-2">
   295→            <ShieldAlert className="w-3 h-3" />
   296→            <span>{t('nav_approvals')}</span>
   297→          </div>
   298→          <h2 className="text-xl lg:text-2xl font-bold gradient-text mb-3">
   299→            {lang === 'bm' ? 'Papan Pemuka Kelulusan' : 'Approval Dashboard'}
   300→          </h2>
   301→          <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   302→            <StatTile
   303→              icon={<Clock className="w-4 h-4" />}
   304→              label={t('stats_pending')}
   305→              value={stats.totalPending}
   306→              tone="amber"
   307→            />
   308→            <StatTile
   309→              icon={<Calendar className="w-4 h-4" />}
   310→              label={lang === 'bm' ? 'Hari Ini' : "Today's Pending"}
   311→              value={stats.todayPending}
   312→              tone="teal"
   313→            />
   314→            <StatTile
   315→              icon={<ShieldAlert className="w-4 h-4" />}
   316→              label={lang === 'bm' ? 'Kelulusan Berperingkat' : 'Multi-level'}
   317→              value={stats.multiLevel}
   318→              tone="rose"
   319→            />
   320→            <StatTile
   321→              icon={<Inbox className="w-4 h-4" />}
   322→              label={lang === 'bm' ? 'Perlu Maklumat' : 'Needs Info'}
   323→              value={bookings?.filter((b) => b.status === 'NEEDS_INFO').length ?? 0}
   324→              tone="orange"
   325→            />
   326→          </div>
   327→        </div>
   328→      </GlassCard>
   329→
   330→      {/* ===== Search + Tabs ===== */}
   331→      <GlassCard className="p-4">
   332→        <div className="flex flex-col lg:flex-row gap-3 lg:items-center">
   333→          <Tabs value={tab} onValueChange={(v) => setTab(v as 'PENDING' | 'NEEDS_INFO')}>
   334→            <TabsList className="glass-input">
   335→              <TabsTrigger value="PENDING" className="text-xs">
   336→                {t('status_PENDING')}
   337→              </TabsTrigger>
   338→              <TabsTrigger value="NEEDS_INFO" className="text-xs">
   339→                {t('status_NEEDS_INFO')}
   340→              </TabsTrigger>
   341→            </TabsList>
   342→          </Tabs>
   343→          <div className="relative flex-1">
   344→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   345→            <Input
   346→              value={search}
   347→              onChange={(e) => setSearch(e.target.value)}
   348→              placeholder={`${t('search')} — ${t('form_facility')} / ${t('name')} / ${lang === 'bm' ? 'No. Rujukan' : 'Ref No.'}`}
   349→              className="pl-9 glass-input w-full"
   350→            />
   351→          </div>
   352→          <div className="text-xs text-muted-foreground whitespace-nowrap">
   353→            {t('showing')} <span className="font-semibold text-foreground">{filtered.length}</span> {t('of')} <span className="font-semibold text-foreground">{bookings?.length ?? 0}</span> {t('records')}
   354→          </div>
   355→        </div>
   356→      </GlassCard>
   357→
   358→      {/* ===== Two-column layout ===== */}
   359→      <div className="grid grid-cols-1 lg:grid-cols-12 gap-4">
   360→        {/* ---- Left: list ---- */}
   361→        <div className="lg:col-span-5 xl:col-span-4">
   362→          <GlassCard className="p-3 h-full">
   363→            <div className="flex items-center justify-between px-2 pb-2 mb-2 border-b border-border/40">
   364→              <h3 className="text-sm font-semibold flex items-center gap-2">
   365→                <Clock className="w-4 h-4 text-amber-500" />
   366→                {tab === 'PENDING' ? t('status_PENDING') : t('status_NEEDS_INFO')}
   367→              </h3>
   368→              <span className="text-[10px] text-muted-foreground">{filtered.length}</span>
   369→            </div>
   370→            <div className="max-h-[640px] overflow-y-auto scroll-area-thin space-y-2 pr-1">
   371→              {isLoading ? (
   372→                Array.from({ length: 4 }).map((_, i) => (
   373→                  <Skeleton key={i} className="h-24 w-full rounded-xl" />
   374→                ))
   375→              ) : filtered.length === 0 ? (
   376→                <EmptyState lang={lang} />
   377→              ) : (
   378→                filtered.map((b) => (
   379→                  <ApprovalListItem
   380→                    key={b.id}
   381→                    booking={b}
   382→                    selected={selected?.id === b.id}
   383→                    onSelect={() => setSelectedId(b.id)}
   384→                  />
   385→                ))
   386→              )}
   387→            </div>
   388→          </GlassCard>
   389→        </div>
   390→
   391→        {/* ---- Right: detail ---- */}
   392→        <div className="lg:col-span-7 xl:col-span-8">
   393→          {isLoading ? (
   394→            <GlassCard className="p-6">
   395→              <Skeleton className="h-8 w-1/3 mb-4" />
   396→              <div className="space-y-3">
   397→                <Skeleton className="h-20 w-full" />
   398→                <Skeleton className="h-32 w-full" />
   399→                <Skeleton className="h-24 w-full" />
   400→              </div>
   401→            </GlassCard>
   402→          ) : !selected ? (
   403→            <GlassCard className="p-10">
   404→              <EmptyState lang={lang} large />
   405→            </GlassCard>
   406→          ) : (
   407→            <DetailPanel
   408→              booking={selected}
   409→              conflict={conflict}
   410→              conflictLoading={conflictLoading}
   411→              overCapacity={overCapacity}
   412→              onAction={openAction}
   413→              pending={actionMut.isPending}
   414→              lang={lang}
   415→              t={t}
   416→            />
   417→          )}
   418→        </div>
   419→      </div>
   420→
   421→      {/* ===== Action Dialog ===== */}
   422→      <Dialog
   423→        open={!!dialogAction}
   424→        onOpenChange={(o) => {
   425→          if (!o) {
   426→            setDialogAction(null);
   427→            setComment('');
   428→          }
   429→        }}
   430→      >
   431→        <DialogContent className="glass-strong">
   432→          <DialogHeader>
   433→            <DialogTitle className="flex items-center gap-2">
   434→              {dialogAction === 'APPROVE' && (
   435→                <CheckCircle2 className="w-5 h-5 text-emerald-600" />
   436→              )}
   437→              {dialogAction === 'REJECT' && <XCircle className="w-5 h-5 text-rose-600" />}
   438→              {dialogAction === 'REQUEST_INFO' && (
   439→                <MessageSquare className="w-5 h-5 text-amber-600" />
   440→              )}
   441→              {dialogAction === 'APPROVE'
   442→                ? t('approve')
   443→                : dialogAction === 'REJECT'
   444→                ? t('reject')
   445→                : t('request_info')}
   446→            </DialogTitle>
   447→          </DialogHeader>
   448→
   449→          <div className="space-y-3">
   450→            {dialogAction === 'APPROVE' && (
   451→              <Alert className="border-emerald-300/60 bg-emerald-50/70 text-emerald-800">
   452→                <CheckCircle2 className="w-4 h-4" />
   453→                <AlertDescription>
   454→                  {lang === 'bm'
   455→                    ? 'Anda akan meluluskan permohonan ini. Notifikasi akan dihantar kepada pemohon.'
   456→                    : 'You are about to approve this booking. The applicant will be notified.'}
   457→                </AlertDescription>
   458→              </Alert>
   459→            )}
   460→            {dialogAction === 'REJECT' && (
   461→              <Alert variant="destructive">
   462→                <AlertTriangle className="w-4 h-4" />
   463→                <AlertDescription>
   464→                  {lang === 'bm'
   465→                    ? 'Permohonan akan ditolak. Sebab penolakan diperlukan.'
   466→                    : 'This booking will be rejected. A reason is required.'}
   467→                </AlertDescription>
   468→              </Alert>
   469→            )}
   470→            {dialogAction === 'REQUEST_INFO' && (
   471→              <Alert className="border-amber-300/60 bg-amber-50/70 text-amber-800">
   472→                <AlertCircle className="w-4 h-4" />
   473→                <AlertDescription>
   474→                  {lang === 'bm'
   475→                    ? 'Pemohon perlu memberikan maklumat tambahan sebelum kelulusan.'
   476→                    : 'The applicant must provide additional information before approval.'}
   477→                </AlertDescription>
   478→              </Alert>
   479→            )}
   480→
   481→            <div className="space-y-2">
   482→              <Label htmlFor="admin-comment" className="text-xs font-semibold">
   483→                {dialogAction === 'REJECT' ? t('rejection_reason') : t('admin_comment')}
   484→                {(dialogAction === 'REJECT' || dialogAction === 'REQUEST_INFO') && (
   485→                  <span className="text-rose-500 ml-0.5">*</span>
   486→                )}
   487→              </Label>
   488→              <Textarea
   489→                id="admin-comment"
   490→                value={comment}
   491→                onChange={(e) => setComment(e.target.value)}
   492→                rows={4}
   493→                placeholder={
   494→                  dialogAction === 'REJECT'
   495→                    ? lang === 'bm'
   496→                      ? 'Nyatakan sebab penolakan...'
   497→                      : 'State the reason for rejection...'
   498→                    : dialogAction === 'REQUEST_INFO'
   499→                    ? lang === 'bm'
   500→                      ? 'Nyatakan maklumat tambahan yang diperlukan...'
   501→                      : 'Specify the additional info needed...'
   502→                    : lang === 'bm'
   503→                    ? 'Catatan tambahan (pilihan)...'
   504→                    : 'Additional note (optional)...'
   505→                }
   506→                className="glass-input resize-none"
   507→                maxLength={1000}
   508→              />
   509→              <p className="text-[10px] text-muted-foreground text-right">{comment.length}/1000</p>
   510→            </div>
   511→          </div>
   512→
   513→          <DialogFooter>
   514→            <Button
   515→              variant="outline"
   516→              onClick={() => {
   517→                setDialogAction(null);
   518→                setComment('');
   519→              }}
   520→              disabled={actionMut.isPending}
   521→            >
   522→              {t('close')}
   523→            </Button>
   524→            <Button
   525→              onClick={submitAction}
   526→              disabled={
   527→                actionMut.isPending ||
   528→                ((dialogAction === 'REJECT' || dialogAction === 'REQUEST_INFO') && !comment.trim())
   529→              }
   530→              className={cn(
   531→                dialogAction === 'APPROVE' && 'bg-emerald-600 hover:bg-emerald-700 text-white',
   532→                dialogAction === 'REJECT' && 'bg-rose-600 hover:bg-rose-700 text-white',
   533→                dialogAction === 'REQUEST_INFO' && 'bg-amber-600 hover:bg-amber-700 text-white',
   534→              )}
   535→            >
   536→              {actionMut.isPending ? t('loading') : t('send')}
   537→            </Button>
   538→          </DialogFooter>
   539→        </DialogContent>
   540→      </Dialog>
   541→    </motion.div>
   542→  );
   543→}

   545→// ===================== Sub-components =====================
   546→
   547→function StatTile({
   548→  icon,
   549→  label,
   550→  value,
   551→  tone,
   552→}: {
   553→  icon: React.ReactNode;
   554→  label: string;
   555→  value: number;
   556→  tone: 'amber' | 'teal' | 'rose' | 'orange';
   557→}) {
