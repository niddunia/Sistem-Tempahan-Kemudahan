    97→    97→const safeParseEquip = (raw?: string | null): string[] => {
    98→    98→  if (!raw) return [];
    99→    99→  try {
   100→   100→    const v = JSON.parse(raw);
   101→   101→    if (Array.isArray(v)) return v.filter((x) => typeof x === 'string');
   102→   102→    if (typeof v === 'string') return [v];
   103→   103→  } catch {
   104→   104→    return raw.split(',').map((s) => s.trim()).filter(Boolean);
   105→   105→  }
   106→   106→  return [];
   107→   107→};
   108→   108→
   109→   109→const fmtDate = (iso: string, lang: 'bm' | 'en') =>
   110→   110→  new Date(iso).toLocaleDateString(lang === 'bm' ? 'ms-MY' : 'en-GB', {
   111→   111→    weekday: 'short',
   112→   112→    day: 'numeric',
   113→   113→    month: 'short',
   114→   114→    year: 'numeric',
   115→   115→  });
   116→   116→
   117→   117→const fmtDateTime = (iso: string, lang: 'bm' | 'en') =>
   118→   118→  new Date(iso).toLocaleString(lang === 'bm' ? 'ms-MY' : 'en-GB', {
   119→   119→    day: 'numeric',
   120→   120→    month: 'short',
   121→   121→    year: 'numeric',
   122→   122→    hour: '2-digit',
   123→   123→    minute: '2-digit',
   124→   124→  });
   125→   125→
   126→   126→const isSameDay = (iso: string, ref: Date) => {
   127→   127→  const d = new Date(iso);
   128→   128→  return (
   129→   129→    d.getFullYear() === ref.getFullYear() &&
   130→   130→    d.getMonth() === ref.getMonth() &&
   131→   131→    d.getDate() === ref.getDate()
   132→   132→  );
   133→   133→};

   135→   135→// ===================== Main View =====================
   136→   136→export function ApprovalsView() {
   137→   137→  const { t, lang } = useT();
   138→   138→  useApp(); // ensure store ready / language reactive
   139→   139→  const { user } = useCurrentUser();
   140→   140→  const qc = useQueryClient();
   141→   141→
   142→   142→  const [tab, setTab] = useState<'PENDING' | 'NEEDS_INFO'>('PENDING');
   143→   143→  const [search, setSearch] = useState('');
   144→   144→  const [selectedId, setSelectedId] = useState<string | null>(null);
   145→   145→  const [dialogAction, setDialogAction] = useState<ActionKind | null>(null);
   146→   146→  const [comment, setComment] = useState('');
   147→   147→
   148→   148→  // ---- Fetch pending / needs-info approvals ----
   149→   149→  const { data: bookings, isLoading } = useQuery<Booking[]>({
   150→   150→    queryKey: ['approvals', tab],
   151→   151→    queryFn: async () => {
   152→   152→      const r = await fetch(`/api/approvals?status=${tab}`, { credentials: 'include' });
   153→   153→      if (!r.ok) throw new Error('failed');
   154→   154→      const j = await r.json();
   155→   155→      return j.data as Booking[];
   156→   156→    },
   157→   157→    enabled: !!user,
   158→   158→  });
   159→   159→
   160→   160→  // ---- Filter list (search by bookingRef / applicant / facility) ----
   161→   161→  const filtered = useMemo(() => {
   162→   162→    if (!bookings) return [];
   163→   163→    const s = search.trim().toLowerCase();
   164→   164→    if (!s) return bookings;
   165→   165→    return bookings.filter((b) =>
   166→   166→      b.bookingRef.toLowerCase().includes(s) ||
   167→   167→      b.user.name.toLowerCase().includes(s) ||
   168→   168→      b.facility.name.toLowerCase().includes(s),
   169→   169→    );
   170→   170→  }, [bookings, search]);
   171→   171→
   172→   172→  // ---- Stats ----
   173→   173→  const stats = useMemo(() => {
   174→   174→    const pending = bookings?.filter((b) => b.status === 'PENDING') ?? [];
   175→   175→    const today = new Date();
   176→   176→    const todayPending = pending.filter((b) => isSameDay(b.eventDate, today)).length;
   177→   177→    const multiLevel = pending.filter((b) => b.requiresMultiLevel).length;
   178→   178→    return {
   179→   179→      totalPending: pending.length || (tab === 'PENDING' ? bookings?.length ?? 0 : 0),
   180→   180→      todayPending,
   181→   181→      multiLevel,
   182→   182→    };
   183→   183→  }, [bookings, tab]);
   184→   184→
   185→   185→  // ---- Selected booking object (derived; falls back to first in list) ----
   186→   186→  const selected = useMemo(
   187→   187→    () => filtered.find((b) => b.id === selectedId) ?? filtered[0] ?? null,
   188→   188→    [filtered, selectedId],
   189→   189→  );
   190→   190→
   191→   191→  // Track the effective selected id for downstream keys/queries
   192→   192→  const effectiveSelectedId = selected?.id ?? null;
   193→   193→
   194→   194→  // ---- Conflict check for selected booking ----
   195→   195→  const { data: conflict, isLoading: conflictLoading } = useQuery<ConflictResult>({
   196→   196→    queryKey: ['conflict-check', effectiveSelectedId],
   197→   197→    queryFn: async () => {
   198→   198→      if (!selected) return { hasConflict: false, conflictingBookings: [] };
   199→   199→      const r = await fetch('/api/bookings/check-conflict', {
   200→   200→        method: 'POST',
   201→   201→        headers: { 'Content-Type': 'application/json' },
   202→   202→        credentials: 'include',
   203→   203→        body: JSON.stringify({
   204→   204→          facilityId: selected.facility.id,
   205→   205→          eventDate: selected.eventDate,
   206→   206→          startTime: selected.startTime,
   207→   207→          endTime: selected.endTime,
   208→   208→          excludeBookingId: selected.id,
   209→   209→        }),
   210→   210→      });
   211→   211→      if (!r.ok) throw new Error('conflict-check failed');
   212→   212→      const j = await r.json();
   213→   213→      return j.data as ConflictResult;
   214→   214→    },
   215→   215→    enabled: !!selected,
   216→   216→  });
   217→   217→
   218→   218→  // ---- Action mutation ----
   219→   219→  const actionMut = useMutation({
   220→   220→    mutationFn: async (vars: { id: string; action: ActionKind; comment: string }) => {
   221→   221→      const r = await fetch(`/api/approvals/${vars.id}`, {
   222→   222→        method: 'PATCH',
   223→   223→        headers: { 'Content-Type': 'application/json' },
   224→   224→        credentials: 'include',
   225→   225→        body: JSON.stringify({ action: vars.action, comment: vars.comment || undefined }),
   226→   226→      });
   227→   227→      if (r.status === 409) {
   228→   228→        const j = await r.json().catch(() => ({}));
   229→   229→        throw new Error(j.message ?? (lang === 'bm' ? 'Pertindihan dikesan.' : 'Conflict detected.'));
   230→   230→      }
   231→   231→      if (!r.ok) {
   232→   232→        const j = await r.json().catch(() => ({}));
   233→   233→        throw new Error(j.message ?? (lang === 'bm' ? 'Tindakan gagal.' : 'Action failed.'));
   234→   234→      }
   235→   235→      return r.json();
   236→   236→    },
   237→   237→    onSuccess: (_data, vars) => {
   238→   238→      const msg =
   239→   239→        vars.action === 'APPROVE'
   240→   240→          ? lang === 'bm' ? 'Permohonan diluluskan.' : 'Booking approved.'
   241→   241→          : vars.action === 'REJECT'
   242→   242→          ? lang === 'bm' ? 'Permohonan ditolak.' : 'Booking rejected.'
   243→   243→          : lang === 'bm' ? 'Permintaan maklumat dihantar.' : 'Info request sent.';
   244→   244→      toast.success(msg);
   245→   245→      // refresh both lists & related caches
   246→   246→      qc.invalidateQueries({ queryKey: ['approvals'] });
   247→   247→      qc.invalidateQueries({ queryKey: ['conflict-check'] });
   248→   248→      qc.invalidateQueries({ queryKey: ['notifications'] });
   249→   249→      // auto-select next pending item (clear current selection so the
   250→   250→      // derived `selected` falls back to the next item in the refreshed list)
   251→   251→      setSelectedId((prev) => (prev === vars.id ? null : prev));
   252→   252→    },
   253→   253→    onError: (err: unknown) => {
   254→   254→      const msg = err instanceof Error ? err.message : 'Error';
   255→   255→      toast.error(msg);
   256→   256→    },
   257→   257→    onSettled: () => {
   258→   258→      setDialogAction(null);
   259→   259→      setComment('');
   260→   260→    },
   261→   261→  });
   262→   262→
   263→   263→  // ---- Open dialog with action ----
   264→   264→  const openAction = (a: ActionKind) => {
   265→   265→    setComment('');
   266→   266→    setDialogAction(a);
   267→   267→  };
   268→   268→
   269→   269→  // ---- Submit dialog ----
   270→   270→  const submitAction = () => {
   271→   271→    if (!selected || !dialogAction) return;

   273→   273→      toast.error(lang === 'bm' ? 'Sila isi ruangan ulasan.' : 'Please fill in the comment.');
   274→   274→      return;
   275→   275→    }
   276→   276→    actionMut.mutate({ id: selected.id, action: dialogAction, comment: comment.trim() });
   277→   277→  };
   278→   278→
   279→   279→  // ---- Capacity check ----
   280→   280→  const overCapacity = selected ? selected.participantCount > selected.facility.capacity : false;
   281→   281→
   282→   282→  return (
   283→   283→    <motion.div
   284→   284→      initial={{ opacity: 0, y: 10 }}
   285→   285→      animate={{ opacity: 1, y: 0 }}
   286→   286→      transition={{ duration: 0.3 }}
   287→   287→      className="space-y-4"
   288→   288→    >
   289→   289→      {/* ===== Header / Stats ===== */}
   290→   290→      <GlassCard className="p-5 lg:p-6 relative overflow-hidden">
   291→   291→        <div className="absolute top-0 right-0 w-64 h-64 rounded-full bg-teal-500/10 blur-3xl pointer-events-none" />
   292→   292→        <div className="absolute -bottom-12 -left-12 w-48 h-48 rounded-full bg-amber-500/10 blur-3xl pointer-events-none" />
   293→   293→        <div className="relative z-10">
   294→   294→          <div className="flex items-center gap-2 text-[11px] text-muted-foreground uppercase tracking-wider mb-2">
   295→   295→            <ShieldAlert className="w-3 h-3" />
   296→   296→            <span>{t('nav_approvals')}</span>
   297→   297→          </div>
   298→   298→          <h2 className="text-xl lg:text-2xl font-bold gradient-text mb-3">
   299→   299→            {lang === 'bm' ? 'Papan Pemuka Kelulusan' : 'Approval Dashboard'}
   300→   300→          </h2>
   301→   301→          <div className="grid grid-cols-2 lg:grid-cols-4 gap-3">
   302→   302→            <StatTile
   303→   303→              icon={<Clock className="w-4 h-4" />}
   304→   304→              label={t('stats_pending')}
   305→   305→              value={stats.totalPending}
   306→   306→              tone="amber"
   307→   307→            />
   308→   308→            <StatTile
   309→   309→              icon={<Calendar className="w-4 h-4" />}
   310→   310→              label={lang === 'bm' ? 'Hari Ini' : "Today's Pending"}
   311→   311→              value={stats.todayPending}
   312→   312→              tone="teal"
   313→   313→            />
   314→   314→            <StatTile
   315→   315→              icon={<ShieldAlert className="w-4 h-4" />}
   316→   316→              label={lang === 'bm' ? 'Kelulusan Berperingkat' : 'Multi-level'}
   317→   317→              value={stats.multiLevel}
   318→   318→              tone="rose"
   319→   319→            />
   320→   320→            <StatTile
   321→   321→              icon={<Inbox className="w-4 h-4" />}
   322→   322→              label={lang === 'bm' ? 'Perlu Maklumat' : 'Needs Info'}
   323→   323→              value={bookings?.filter((b) => b.status === 'NEEDS_INFO').length ?? 0}
   324→   324→              tone="orange"
   325→   325→            />
   326→   326→          </div>
   327→   327→        </div>
   328→   328→      </GlassCard>
   329→   329→
   330→   330→      {/* ===== Search + Tabs ===== */}
   331→   331→      <GlassCard className="p-4">
   332→   332→        <div className="flex flex-col lg:flex-row gap-3 lg:items-center">
   333→   333→          <Tabs value={tab} onValueChange={(v) => setTab(v as 'PENDING' | 'NEEDS_INFO')}>
   334→   334→            <TabsList className="glass-input">
   335→   335→              <TabsTrigger value="PENDING" className="text-xs">
   336→   336→                {t('status_PENDING')}
   337→   337→              </TabsTrigger>
   338→   338→              <TabsTrigger value="NEEDS_INFO" className="text-xs">
   339→   339→                {t('status_NEEDS_INFO')}
   340→   340→              </TabsTrigger>
   341→   341→            </TabsList>
   342→   342→          </Tabs>
   343→   343→          <div className="relative flex-1">
   344→   344→            <Search className="absolute left-3 top-1/2 -translate-y-1/2 w-4 h-4 text-muted-foreground" />
   345→   345→            <Input
   346→   346→              value={search}
   347→   347→              onChange={(e) => setSearch(e.target.value)}
   348→   348→              placeholder={`${t('search')} — ${t('form_facility')} / ${t('name')} / ${lang === 'bm' ? 'No. Rujukan' : 'Ref No.'}`}
   349→   349→              className="pl-9 glass-input w-full"
   350→   350→            />
   351→   351→          </div>
   352→   352→          <div className="text-xs text-muted-foreground whitespace-nowrap">
   353→   353→            {t('showing')} <span className="font-semibold text-foreground">{filtered.length}</span> {t('of')} <span className="font-semibold text-foreground">{bookings?.length ?? 0}</span> {t('records')}
   354→   354→          </div>
   355→   355→        </div>
   356→   356→      </GlassCard>
   357→   357→
   358→   358→      {/* ===== Two-column layout ===== */}
   359→   359→      <div className="grid grid-cols-1 lg:grid-cols-12 gap-4">
   360→   360→        {/* ---- Left: list ---- */}
   361→   361→        <div className="lg:col-span-5 xl:col-span-4">

   364→   364→              <h3 className="text-sm font-semibold flex items-center gap-2">

   367→   367→              </h3>
   368→   368→              <span className="text-[10px] text-muted-foreground">{filtered.length}</span>

   371→   371→              {isLoading ? (

   392→   392→        <div className="lg:col-span-7 xl:col-span-8">

   418→   418→        </div>
   419→   419→      </div>

   421→   421→      {/* ===== Action Dialog ===== */}
   422→   422→      <Dialog

   430→   430→      >
   431→   431→        <DialogContent className="glass-strong">
   432→   432→          <DialogHeader>

   448→   448→
   449→   449→          <div className="space-y-3">

   453→   453→                <AlertDescription>
