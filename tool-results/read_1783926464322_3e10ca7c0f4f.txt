    89→    89→
    90→    90→export function BookingFormView() {
    91→    91→  const { t, lang } = useT();
    92→    92→  const { bookingPrefill, setBookingPrefill, setView } = useApp();
    93→    93→
    94→    94→  /* ---------------- form state ---------------- */
    95→    95→  const [category, setCategory] = useState<CategoryFilter>('all');
    96→    96→  const [facilityId, setFacilityId] = useState('');
    97→    97→  const [date, setDate] = useState<Date | undefined>();
    98→    98→  const [startTime, setStartTime] = useState('');
    99→    99→  const [endTime, setEndTime] = useState('');
   100→   100→  const [purpose, setPurpose] = useState('');
   101→   101→  const [programName, setProgramName] = useState('');
   102→   102→  const [participantCount, setParticipantCount] = useState('');
   103→   103→  const [equipmentNeeded, setEquipmentNeeded] = useState('');
   104→   104→  const [notes, setNotes] = useState('');
   105→   105→
   106→   106→  const [submitting, setSubmitting] = useState(false);
   107→   107→  const [savingDraft, setSavingDraft] = useState(false);
   108→   108→  const [checkingConflict, setCheckingConflict] = useState(false);
   109→   109→  const [conflictResult, setConflictResult] = useState<ConflictResult | null>(null);
   110→   110→  const [validated, setValidated] = useState(false);
   111→   111→  const [prefillApplied, setPrefillApplied] = useState(false);
   112→   112→
   113→   113→  /* ---------------- fetch facilities ---------------- */
   114→   114→  const { data: facilities, isLoading: loadingFacilities } = useQuery<Facility[]>({
   115→   115→    queryKey: ['facilities'],
   116→   116→    queryFn: async () => {
   117→   117→      const r = await fetch('/api/facilities');
   118→   118→      const j = await r.json();
   119→   119→      return j.data as Facility[];
   120→   120→    },
   121→   121→  });
   122→   122→
   123→   123→  const filteredFacilities = useMemo(() => {
   124→   124→    if (!facilities) return [];
   125→   125→    if (category === 'all') return facilities;
   126→   126→    return facilities.filter((f) => f.category === category);
   127→   127→  }, [facilities, category]);
   128→   128→
   129→   129→  const selectedFacility = useMemo(
   130→   130→    () => facilities?.find((f) => f.id === facilityId) ?? null,
   131→   131→    [facilities, facilityId],
   132→   132→  );
   133→   133→
   134→   134→  /* ---------------- pre-fill from AI parse ---------------- */
   135→   135→  useEffect(() => {
   136→   136→    if (prefillApplied || !bookingPrefill) return;
   137→   137→    const p = bookingPrefill;
   138→   138→    if (typeof p.facilityId === 'string' && p.facilityId) {
   139→   139→      setFacilityId(p.facilityId);
   140→   140→      const f = facilities?.find((x) => x.id === p.facilityId);
   141→   141→      if (f) setCategory(f.category as CategoryFilter);
   142→   142→    }
   143→   143→    if (typeof p.date === 'string' && p.date) {
   144→   144→      const d = new Date(p.date);
   145→   145→      if (!isNaN(d.getTime())) {
   146→   146→        d.setHours(0, 0, 0, 0);
   147→   147→        setDate(d);
   148→   148→      }
   149→   149→    }
   150→   150→    if (typeof p.startTime === 'string' && p.startTime) setStartTime(p.startTime);
   151→   151→    if (typeof p.endTime === 'string' && p.endTime) setEndTime(p.endTime);
   152→   152→    if (typeof p.participants === 'number' && p.participants > 0) {
   153→   153→      setParticipantCount(String(p.participants));
   154→   154→    }
   155→   155→    if (typeof p.purpose === 'string' && p.purpose) setPurpose(p.purpose);
   156→   156→    setPrefillApplied(true);
   157→   157→    setBookingPrefill(null);
   158→   158→    toast.info(
   159→   159→      lang === 'bm'
   160→   160→        ? 'Borang diisi automatik dari AI'
   161→   161→        : 'Form auto-filled from AI',
   162→   162→    );
   163→   163→  }, [bookingPrefill, facilities, prefillApplied, setBookingPrefill, lang]);
   164→   164→
   165→   165→  /* ---------------- time options ---------------- */
   166→   166→  const timeOptions = useMemo(() => {
   167→   167→    const start = selectedFacility?.operatingStart ?? '08:00';
   168→   168→    const end = selectedFacility?.operatingEnd ?? '22:00';
   169→   169→    const startMin = parseInt(start.slice(0, 2), 10) * 60 + parseInt(start.slice(3, 5), 10);
   170→   170→    const endMin = parseInt(end.slice(0, 2), 10) * 60 + parseInt(end.slice(3, 5), 10);
   171→   171→    const opts: { value: string; label: string }[] = [];
   172→   172→    for (let m = startMin; m < endMin; m += 30) {
   173→   173→      const h = Math.floor(m / 60).toString().padStart(2, '0');
   174→   174→      const min = (m % 60).toString().padStart(2, '0');
   175→   175→      opts.push({ value: `${h}:${min}`, label: `${h}:${min}` });
   176→   176→    }
   177→   177→    return opts;
   178→   178→  }, [selectedFacility]);
   179→   179→
   180→   180→  /* ---------------- derived: capacity + time validity ---------------- */
   181→   181→  const participants = parseInt(participantCount, 10);
   182→   182→  const capacityExceeded =
   183→   183→    !!selectedFacility && !isNaN(participants) && participants > selectedFacility.capacity;
   184→   184→  const invalidTime = !!startTime && !!endTime && startTime >= endTime;
   185→   185→
   186→   186→  /* ---------------- real-time conflict check (debounced) ---------------- */
   187→   187→  useEffect(() => {
   188→   188→    if (!facilityId || !date || !startTime || !endTime || invalidTime) {
   189→   189→      setConflictResult(null);
   190→   190→      setCheckingConflict(false);
   191→   191→      return;
   192→   192→    }
   193→   193→    let cancelled = false;
   194→   194→    setCheckingConflict(true);
   195→   195→    const iso = new Date(date);
   196→   196→    iso.setHours(0, 0, 0, 0);
   197→   197→    const timer = setTimeout(async () => {
   198→   198→      try {
   199→   199→        const r = await fetch('/api/bookings/check-conflict', {
   200→   200→          method: 'POST',
   201→   201→          headers: { 'Content-Type': 'application/json' },
   202→   202→          body: JSON.stringify({
   203→   203→            facilityId,
   204→   204→            eventDate: iso.toISOString(),
   205→   205→            startTime,
   206→   206→            endTime,
   207→   207→          }),
   208→   208→        });
   209→   209→        const j = await r.json();
   210→   210→        if (!cancelled && j?.data) setConflictResult(j.data as ConflictResult);
   211→   211→      } catch {
   212→   212→        /* network error — silent */
   213→   213→      } finally {
   214→   214→        if (!cancelled) setCheckingConflict(false);
   215→   215→      }
   216→   216→    }, 450);
   217→   217→    return () => {
   218→   218→      cancelled = true;
   219→   219→      clearTimeout(timer);
   220→   220→    };
   221→   221→  }, [facilityId, date, startTime, endTime, invalidTime]);
   222→   222→
   223→   223→  /* ---------------- validation ---------------- */
   224→   224→  const errors = useMemo<Record<string, string>>(() => {
   225→   225→    const e: Record<string, string> = {};
   226→   226→    if (!facilityId) e.facilityId = lang === 'bm' ? 'Sila pilih kemudahan' : 'Please select a facility';
   227→   227→    if (!date) e.date = lang === 'bm' ? 'Sila pilih tarikh' : 'Please select a date';
   228→   228→    if (!startTime) e.startTime = lang === 'bm' ? 'Sila pilih masa mula' : 'Please select start time';
   229→   229→    if (!endTime) e.endTime = lang === 'bm' ? 'Sila pilih masa tamat' : 'Please select end time';
   230→   230→    if (invalidTime) e.time = lang === 'bm' ? 'Masa tamat mesti selepas masa mula' : 'End time must be after start time';
   231→   231→    if (!purpose.trim() || purpose.trim().length < 3)
   232→   232→      e.purpose = lang === 'bm' ? 'Tujuan diperlukan (minimum 3 aksara)' : 'Purpose required (min 3 chars)';
   233→   233→    if (!participantCount || isNaN(participants) || participants < 1)
   234→   234→      e.participants = lang === 'bm' ? 'Bilangan peserta diperlukan' : 'Participant count required';
   235→   235→    return e;
   236→   236→  }, [facilityId, date, startTime, endTime, purpose, participantCount, participants, invalidTime, lang]);
   237→   237→
   238→   238→  const hasErrors = Object.keys(errors).length > 0;
   239→   239→  const showErrors = validated;
   240→   240→  const hasConflict = !!conflictResult?.hasConflict;
   241→   241→  const canSubmit = !hasErrors && !capacityExceeded && !hasConflict && !checkingConflict;
   242→   242→
   243→   243→  /* ---------------- alternative slot click ---------------- */
   244→   244→  const applyAlternative = (alt: ConflictAlternative) => {
   245→   245→    setStartTime(alt.startTime);
   246→   246→    setEndTime(alt.endTime);
   247→   247→    toast.success(
   248→   248→      lang === 'bm'
   249→   249→        ? `Slot alternatif dipilih: ${alt.startTime} – ${alt.endTime}`
   250→   250→        : `Alternative slot applied: ${alt.startTime} – ${alt.endTime}`,
   251→   251→    );
   252→   252→  };
   253→   253→
   254→   254→  /* ---------------- build payload helper ---------------- */
   255→   255→  const buildPayload = (asDraft: boolean) => {
   256→   256→    const iso = date ? new Date(date) : new Date();
   257→   257→    iso.setHours(0, 0, 0, 0);
   258→   258→    return {
   259→   259→      facilityId,
   260→   260→      eventDate: iso.toISOString(),
   261→   261→      startTime: startTime || (selectedFacility?.operatingStart ?? '08:00'),
   262→   262→      endTime: endTime || (selectedFacility?.operatingStart ?? '10:00'),
   263→   263→      purpose: purpose.trim() || (lang === 'bm' ? 'Draf Permohonan' : 'Draft Booking'),
   264→   264→      programName: programName.trim() || undefined,
   265→   265→      participantCount: !isNaN(participants) && participants > 0 ? participants : 1,
   266→   266→      equipmentNeeded: equipmentNeeded.trim() || undefined,
   267→   267→      notes: notes.trim() || undefined,
   268→   268→      isDraft: asDraft,
   269→   269→    };
   270→   270→  };
   271→   271→
   272→   272→  /* ---------------- submit handler ---------------- */
   273→   273→  const handleSubmit = async (asDraft: boolean) => {
   274→   274→    setValidated(true);
   275→   275→    if (hasErrors) {
   276→   276→      toast.error(lang === 'bm' ? 'Sila lengkapkan borang' : 'Please complete the form');
   277→   277→      return;
   278→   278→    }
   279→   279→    if (asDraft) {
   280→   280→      setSavingDraft(true);
   281→   281→    } else {
   282→   282→      if (capacityExceeded) {
   283→   283→        toast.error(t('form_capacity_warn'));
   284→   284→        return;
   285→   285→      }
   286→   286→      setSubmitting(true);
   287→   287→      // Real-time conflict re-check before submission
   288→   288→      try {
   289→   289→        const iso = new Date(date!);
   290→   290→        iso.setHours(0, 0, 0, 0);
   291→   291→        const checkR = await fetch('/api/bookings/check-conflict', {
   292→   292→          method: 'POST',
   293→   293→          headers: { 'Content-Type': 'application/json' },
   294→   294→          body: JSON.stringify({
   295→   295→            facilityId,
   296→   296→            eventDate: iso.toISOString(),
   297→   297→            startTime,
   298→   298→            endTime,
   299→   299→          }),
   300→   300→        });
   301→   301→        const checkJ = await checkR.json();
   302→   302→        if (checkJ?.data?.hasConflict) {
   303→   303→          setConflictResult(checkJ.data as ConflictResult);
   304→   304→          toast.error(t('form_conflict_warn'));
   305→   305→          setSubmitting(false);
   306→   306→          return;
   307→   307→        }
   308→   308→      } catch {
   309→   309→        toast.error(lang === 'bm' ? 'Ralat semakan konflik' : 'Conflict check error');
   310→   310→        setSubmitting(false);
   311→   311→        return;
   312→   312→      }
   313→   313→    }
   314→   314→
   315→   315→    try {
   316→   316→      const payload = buildPayload(asDraft);
   317→   317→      const r = await fetch('/api/bookings', {
   318→   318→        method: 'POST',
   319→   319→        headers: { 'Content-Type': 'application/json' },
   320→   320→        body: JSON.stringify(payload),
   321→   321→      });
   322→   322→      if (!r.ok) {
   323→   323→        const j = await r.json().catch(() => null);
   324→   324→        const msg = j?.message ?? (lang === 'bm' ? 'Permohonan gagal dihantar' : 'Submission failed');
   325→   325→        toast.error(msg);
   326→   326→        return;
   327→   327→      }
   328→   328→      if (asDraft) {
   329→   329→        toast.info(lang === 'bm' ? 'Draf berjaya disimpan.' : 'Draft saved successfully.');
   330→   330→      } else {
   331→   331→        toast.success(t('form_success'));
   332→   332→      }
   333→   333→      setView('dashboard');
   334→   334→    } catch {
   335→   335→      toast.error(lang === 'bm' ? 'Ralat rangkaian. Cuba lagi.' : 'Network error. Try again.');
   336→   336→    } finally {
   337→   337→      setSubmitting(false);
   338→   338→      setSavingDraft(false);
   339→   339→    }
   340→   340→  };
   341→   341→
   342→   342→  /* ---------------- helpers ---------------- */
   343→   343→  const equipmentList = useMemo(() => {
   344→   344→    if (!selectedFacility?.equipment) return [];
   345→   345→    try {
   346→   346→      const parsed = JSON.parse(selectedFacility.equipment);
   347→   347→      return Array.isArray(parsed) ? parsed : [];
   348→   348→    } catch {
   349→   349→      return selectedFacility.equipment
   350→   350→        .split(/[,;]/)
   351→   351→        .map((s) => s.trim())
   352→   352→        .filter(Boolean);
   353→   353→    }
   354→   354→  }, [selectedFacility]);
   355→   355→
   356→   356→  const locale = lang === 'bm' ? 'ms-MY' : 'en-GB';
   357→   357→  const today = new Date();
   358→   358→  today.setHours(0, 0, 0, 0);
   359→   359→
   360→   360→  const hasAlternatives = hasConflict && (conflictResult?.alternatives?.length ?? 0) > 0;
   361→   361→
   362→   362→  /* ---------------- render ---------------- */
   363→   363→  return (
   364→   364→    <motion.div
   365→   365→      initial={{ opacity: 0, y: 10 }}
   366→   366→      animate={{ opacity: 1, y: 0 }}
   367→   367→      transition={{ duration: 0.3 }}
   368→   368→      className="space-y-4 pb-8"
   369→   369→    >
   370→   370→      {/* ===== HERO ===== */}
   371→   371→      <GlassCard className="p-5 lg:p-6 relative overflow-hidden">
   372→   372→        <div className="absolute top-0 right-0 w-64 h-64 rounded-full bg-teal-500/10 blur-3xl pointer-events-none" />
   373→   373→        <div className="absolute -bottom-12 -left-12 w-48 h-48 rounded-full bg-amber-500/10 blur-3xl pointer-events-none" />
   374→   374→        <div className="relative z-10">
   375→   375→          <div className="flex items-center gap-2 text-[11px] text-muted-foreground uppercase tracking-wider mb-2">
   376→   376→            <Sparkles className="w-3 h-3 text-teal-500" />
   377→   377→            <span>{lang === 'bm' ? 'Borang Permohonan' : 'Booking Application'}</span>
   378→   378→          </div>
   379→   379→          <h2 className="text-xl lg:text-2xl font-bold gradient-text mb-1">
   380→   380→            {t('nav_book')}
   381→   381→          </h2>
   382→   382→          <p className="text-sm text-muted-foreground max-w-2xl">
   383→   383→            {lang === 'bm'
   384→   384→              ? 'Isi borang di bawah untuk memohon tempahan kemudahan. Semakan konflik & kapasiti dibuat secara automatik.'
   385→   385→              : 'Complete the form below to apply for a facility booking. Conflict and capacity checks run automatically.'}
   386→   386→          </p>
   387→   387→        </div>
   388→   388→      </GlassCard>
   389→   389→
   390→   390→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-4 min-w-0">
   391→   391→        {/* ===== MAIN FORM (2 cols) ===== */}
   392→   392→        <GlassCard className="lg:col-span-2 p-5 lg:p-6 space-y-5 min-w-0 overflow-visible">
   393→   393→          {/* Category + Facility */}
   394→   394→          <div className="space-y-4">
   395→   395→            <div className="space-y-2">
   396→   396→              <Label className="text-xs font-semibold uppercase tracking-wider text-muted-foreground flex items-center gap-1.5">
   397→   397→                <Tag className="w-3 h-3 text-teal-500" />
   398→   398→                {lang === 'bm' ? 'Jenis Kemudahan' : 'Facility Type'}
   399→   399→              </Label>
   400→   400→              <div className="grid grid-cols-3 gap-2">
   401→   401→                <CategoryChip
   402→   402→                  active={category === 'all'}
   403→   403→                  onClick={() => {
   404→   404→                    setCategory('all');
   405→   405→                    setFacilityId('');
   406→   406→                  }}
   407→   407→                  icon={<Building2 className="w-3.5 h-3.5" />}
   408→   408→                  label={lang === 'bm' ? 'Semua' : 'All'}
   409→   409→                />
   410→   410→                <CategoryChip
   411→   411→                  active={category === 'COMPUTER_ROOM'}
   412→   412→                  onClick={() => {
   413→   413→                    setCategory('COMPUTER_ROOM');
   414→   414→                    setFacilityId('');
   415→   415→                  }}
   416→   416→                  icon={<Monitor className="w-3.5 h-3.5" />}
   417→   417→                  label={lang === 'bm' ? 'Bilik Komputer' : 'Computer Room'}
   418→   418→                />
   419→   419→                <CategoryChip
   420→   420→                  active={category === 'LECTURE_HALL'}
   421→   421→                  onClick={() => {
   422→   422→                    setCategory('LECTURE_HALL');
   423→   423→                    setFacilityId('');
   424→   424→                  }}
   425→   425→                  icon={<GraduationCap className="w-3.5 h-3.5" />}
   426→   426→                  label={lang === 'bm' ? 'Dewan Kuliah' : 'Lecture Hall'}
   427→   427→                />
   428→   428→              </div>
   429→   429→            </div>
   430→   430→
   431→   431→            <div className="space-y-2">
   432→   432→              <Label htmlFor="facility" className="text-xs font-semibold uppercase tracking-wider text-muted-foreground flex items-center gap-1.5">
   433→   433→                <Building2 className="w-3 h-3 text-teal-500" />
   434→   434→                {t('form_facility')} <span className="text-destructive">*</span>
   435→   435→              </Label>
   436→   436→              <Select value={facilityId} onValueChange={setFacilityId}>
   437→   437→                <SelectTrigger
   438→   438→                  id="facility"
   439→   439→                  className={cn('glass-input w-full min-w-0', showErrors && errors.facilityId && 'border-destructive ring-destructive/30')}
   440→   440→                >
   441→   441→                  <SelectValue placeholder={
   442→   442→                    loadingFacilities
