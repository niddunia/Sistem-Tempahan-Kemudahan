    91→    91→/** Convert any CSS color to an rgba string with given alpha. Falls back to the original color if parsing fails. */
    92→    92→function toRgba(color: string, alpha: number): string {
    93→    93→  if (!color) return `rgba(20, 184, 166, ${alpha})`;
    94→    94→  const hex = color.trim().startsWith('#') ? color.trim().slice(1) : color.trim();
    95→    95→  if (/^[0-9a-fA-F]{3}$/.test(hex)) {
    96→    96→    const expanded = hex.split('').map((c) => c + c).join('');
    97→    97→    const r = parseInt(expanded.slice(0, 2), 16);
    98→    98→    const g = parseInt(expanded.slice(2, 4), 16);
    99→    99→    const b = parseInt(expanded.slice(4, 6), 16);
   100→   100→    return `rgba(${r}, ${g}, ${b}, ${alpha})`;
   101→   101→  }
   102→   102→  if (/^[0-9a-fA-F]{6}$/.test(hex)) {
   103→   103→    const r = parseInt(hex.slice(0, 2), 16);
   104→   104→    const g = parseInt(hex.slice(2, 4), 16);
   105→   105→    const b = parseInt(hex.slice(4, 6), 16);
   106→   106→    return `rgba(${r}, ${g}, ${b}, ${alpha})`;
   107→   107→  }
   108→   108→  return color;
   109→   109→}
