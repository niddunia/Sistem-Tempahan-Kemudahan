   109→   109→// ===================== Helpers =====================
   110→   110→const fmtTimestamp = (iso: string, lang: 'bm' | 'en') => {
   111→   111→  try {
   112→   112→    const d = new Date(iso);
   113→   113→    return format(d, lang === 'bm' ? 'dd MMM yyyy, HH:mm:ss' : 'dd MMM yyyy, HH:mm:ss');
   114→   114→  } catch {
   115→   115→    return iso;
   116→   116→  }
   117→   117→};
   118→   118→
   119→   119→const fmtRelative = (iso: string, lang: 'bm' | 'en') => {
   120→   120→  const diff = Date.now() - new Date(iso).getTime();
   121→   121→  const mins = Math.floor(diff / 60_000);
   122→   122→  if (mins < 1) return lang === 'bm' ? 'baru saja' : 'just now';
   123→   123→  if (mins < 60) return lang === 'bm' ? `${mins} minit lalu` : `${mins}m ago`;
   124→   124→  const hrs = Math.floor(mins / 60);
   125→   125→  if (hrs < 24) return lang === 'bm' ? `${hrs} jam lalu` : `${hrs}h ago`;
   126→   126→  const days = Math.floor(hrs / 24);
   127→   127→  return lang === 'bm' ? `${days} hari lalu` : `${days}d ago`;
   128→   128→};
   129→   129→
   130→   130→const parseDetails = (raw: string | null): string | null => {
   131→   131→  if (!raw) return null;
   132→   132→  try {
   133→   133→    return JSON.stringify(JSON.parse(raw), null, 2);
   134→   134→  } catch {
   135→   135→    return raw;
   136→   136→  }
   137→   137→};
   138→   138→
   139→   139→const maskIp = (ip: string | null): string => {
   140→   140→  if (!ip) return '—';
   141→   141→  // Show first two octets for IPv4, mask the rest (PDPA-friendly preview)
   142→   142→  if (/^\d{1,3}(\.\d{1,3}){3}$/.test(ip)) {
   143→   143→    const parts = ip.split('.');
   144→   144→    return `${parts[0]}.${parts[1]}.*.*`;
   145→   145→  }
   146→   146→  return ip;
   147→   147→};
