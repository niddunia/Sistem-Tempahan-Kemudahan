   109→// ===================== Helpers =====================
   110→const fmtTimestamp = (iso: string, lang: 'bm' | 'en') => {
   111→  try {
   112→    const d = new Date(iso);
   113→    return format(d, lang === 'bm' ? 'dd MMM yyyy, HH:mm:ss' : 'dd MMM yyyy, HH:mm:ss');
   114→  } catch {
   115→    return iso;
   116→  }
   117→};
   118→
   119→const fmtRelative = (iso: string, lang: 'bm' | 'en') => {
   120→  const diff = Date.now() - new Date(iso).getTime();
   121→  const mins = Math.floor(diff / 60_000);
   122→  if (mins < 1) return lang === 'bm' ? 'baru saja' : 'just now';
   123→  if (mins < 60) return lang === 'bm' ? `${mins} minit lalu` : `${mins}m ago`;
   124→  const hrs = Math.floor(mins / 60);
   125→  if (hrs < 24) return lang === 'bm' ? `${hrs} jam lalu` : `${hrs}h ago`;
   126→  const days = Math.floor(hrs / 24);
   127→  return lang === 'bm' ? `${days} hari lalu` : `${days}d ago`;
   128→};
   129→
   130→const parseDetails = (raw: string | null): string | null => {
   131→  if (!raw) return null;
   132→  try {
   133→    return JSON.stringify(JSON.parse(raw), null, 2);
   134→  } catch {
   135→    return raw;
   136→  }
   137→};
   138→
   139→const maskIp = (ip: string | null): string => {
   140→  if (!ip) return '—';
   141→  // Show first two octets for IPv4, mask the rest (PDPA-friendly preview)
   142→  if (/^\d{1,3}(\.\d{1,3}){3}$/.test(ip)) {
   143→    const parts = ip.split('.');
   144→    return `${parts[0]}.${parts[1]}.*.*`;
   145→  }
   146→  return ip;
   147→};

   478→// ===================== Audit Row =====================
   479→function AuditRow({

   490→  tr: (bm: string, en: string) => string;
   491→}) {
   492→  const sevCfg = SEVERITY_CONFIG[log.severity] ?? SEVERITY_CONFIG.INFO;
   493→  const SevIcon = sevCfg.icon;
   494→  const moduleCls = MODULE_COLORS[log.module] ?? 'text-zinc-600 bg-zinc-100/70 border-zinc-300/60';
   495→  const prettyDetails = useMemo(() => parseDetails(log.details), [log.details]);
   496→
   497→  return (
   498→    <div
   499→      className={cn(
   500→        'rounded-lg border transition-all',
   501→        expanded
   502→          ? 'border-teal-300/60 bg-teal-50/30 shadow-sm'
   503→          : 'border-border/40 hover:border-teal-200/60 hover:bg-muted/30',
   504→      )}
   505→    >
   506→      {/* Clickable row */}
   507→      <button
   508→        type="button"
   509→        onClick={onToggle}
   510→        aria-expanded={expanded}
   511→        className="w-full text-left px-3 py-3 focus:outline-none focus-visible:ring-2 focus-visible:ring-teal-500/40 rounded-lg"
   512→      >
   513→        {/* Desktop grid layout */}
   514→        <div className="hidden lg:grid grid-cols-[minmax(120px,150px)_minmax(150px,180px)_minmax(100px,120px)_minmax(140px,1fr)_minmax(120px,140px)_minmax(100px,120px)_40px] gap-3 items-center">
   515→          {/* Timestamp */}
   516→          <div className="min-w-0">
   517→            <div className="text-xs font-mono text-foreground truncate">{fmtTimestamp(log.createdAt, lang)}</div>
   518→            <div className="text-[10px] text-muted-foreground">{fmtRelative(log.createdAt, lang)}</div>
   519→          </div>
   520→
   521→          {/* User */}
   522→          <div className="min-w-0 flex items-center gap-1.5">
   523→            {log.user ? (
   524→              <>
   525→                <div className="w-6 h-6 rounded-full bg-gradient-to-br from-teal-400 to-emerald-500 flex items-center justify-center shrink-0">
   526→                  <User className="w-3 h-3 text-white" />
   527→                </div>
   528→                <div className="min-w-0">
   529→                  <div className="text-xs font-medium truncate">{log.user.name}</div>
   530→                  <div className="text-[10px] text-muted-foreground truncate">{log.user.email}</div>
   531→                </div>
   532→              </>
   533→            ) : (
   534→              <div className="flex items-center gap-1.5">
   535→                <div className="w-6 h-6 rounded-full bg-zinc-200 flex items-center justify-center shrink-0">
   536→                  <Cpu className="w-3 h-3 text-zinc-500" />
   537→                </div>
   538→                <span className="text-xs text-muted-foreground italic">{tr('Sistem', 'System')}</span>
   539→              </div>
   540→            )}
   541→          </div>
   542→
   543→          {/* Module badge */}
   544→          <div>
   545→            <span className={cn('inline-flex items-center px-2 py-0.5 rounded-md text-[10px] font-bold border whitespace-nowrap', moduleCls)}>
   546→              {log.module}
   547→            </span>
   548→          </div>
   549→
   550→          {/* Action + severity */}
   551→          <div className="min-w-0 flex items-center gap-2">
   552→            <span className={cn('inline-flex items-center gap-1 px-1.5 py-0.5 rounded text-[10px] font-semibold border shrink-0 whitespace-nowrap', sevCfg.cls)}>
   553→              <SevIcon className="w-2.5 h-2.5" />
   554→            </span>
   555→            <span className="text-xs font-medium whitespace-nowrap font-mono">{log.action}</span>
   556→          </div>
   557→
   558→          {/* Entity */}
   559→          <div className="min-w-0 text-xs">
   560→            {log.entity ? (
   561→              <div className="flex items-center gap-1 min-w-0">
   562→                <span className="text-muted-foreground whitespace-nowrap">{log.entity}</span>
   563→                {log.entityId && (
   564→                  <span className="font-mono text-[10px] text-muted-foreground/70 whitespace-nowrap">#{log.entityId.slice(0, 8)}</span>
   565→                )}
   566→              </div>
   567→            ) : (
   568→              <span className="text-muted-foreground/50">—</span>
   569→            )}
   570→          </div>
   571→
   572→          {/* IP */}
   573→          <div className="text-[11px] font-mono text-muted-foreground whitespace-nowrap">{maskIp(log.ipAddress)}</div>
   574→
   575→          {/* Chevron */}
   576→          <div className="flex justify-end">
   577→            {expanded ? (
   578→              <ChevronUp className="w-4 h-4 text-muted-foreground" />
   579→            ) : (
   580→              <ChevronDown className="w-4 h-4 text-muted-foreground" />
   581→            )}
   582→          </div>
   583→        </div>
   584→
   585→        {/* Mobile stacked layout */}
   586→        <div className="lg:hidden space-y-2">
   587→          <div className="flex items-center justify-between gap-2">
   588→            <div className="flex items-center gap-1.5 flex-wrap">
   589→              <span className={cn('inline-flex items-center px-2 py-0.5 rounded-md text-[10px] font-bold border whitespace-nowrap', moduleCls)}>
   590→                {log.module}
   591→              </span>
   592→              <span className={cn('inline-flex items-center gap-1 px-1.5 py-0.5 rounded text-[10px] font-semibold border whitespace-nowrap', sevCfg.cls)}>
   593→                <SevIcon className="w-2.5 h-2.5" />
   594→                {log.severity}
   595→              </span>
   596→            </div>
   597→            <div className="flex items-center gap-1 text-[10px] text-muted-foreground">
   598→              {fmtRelative(log.createdAt, lang)}
   599→              {expanded ? <ChevronUp className="w-3.5 h-3.5" /> : <ChevronDown className="w-3.5 h-3.5" />}
   600→            </div>
   601→          </div>
   602→
   603→          <div className="text-xs font-mono font-medium break-all">{log.action}</div>
   604→
   605→          <div className="flex items-center justify-between gap-2 text-[11px] text-muted-foreground">
   606→            <div className="flex items-center gap-1.5 min-w-0">
   607→              {log.user ? (
   608→                <>
   609→                  <User className="w-3 h-3 shrink-0" />
   610→                  <span className="truncate">{log.user.name}</span>
   611→                </>
   612→              ) : (
   613→                <>
   614→                  <Cpu className="w-3 h-3 shrink-0" />
   615→                  <span className="italic">{tr('Sistem', 'System')}</span>
   616→                </>
   617→              )}
   618→            </div>
   619→            <span className="font-mono whitespace-nowrap">{maskIp(log.ipAddress)}</span>
   620→          </div>
   621→
   622→          <div className="text-[10px] text-muted-foreground font-mono">
   623→            {fmtTimestamp(log.createdAt, lang)}
   624→          </div>
   625→        </div>
   626→      </button>
   627→
   628→      {/* Expanded details */}
   629→      <AnimatePresence initial={false}>
   630→        {expanded && (
   631→          <motion.div
   632→            initial={{ height: 0, opacity: 0 }}
   633→            animate={{ height: 'auto', opacity: 1 }}
   634→            exit={{ height: 0, opacity: 0 }}
   635→            transition={{ duration: 0.2 }}
   636→            className="overflow-hidden"
   637→          >
   638→            <div className="px-3 pb-3 pt-1 border-t border-border/40 mt-1 space-y-3">
   639→              {/* Meta grid */}
   640→              <div className="grid grid-cols-2 lg:grid-cols-4 gap-3 pt-3">
   641→                <DetailField
   642→                  icon={User}
   643→                  label={tr('Pengguna', 'User')}
   644→                  value={
   645→                    log.user
   646→                      ? `${log.user.name}\n${log.user.email}\n${log.user.role}`
   647→                      : tr('Sistem (tiada pengguna)', 'System (no user)')
   648→                  }
   649→                />
   650→                <DetailField
   651→                  icon={Activity}
   652→                  label={tr('Tindakan', 'Action')}
   653→                  value={log.action}
   654→                  mono
   655→                />
   656→                <DetailField
   657→                  icon={FileText}
   658→                  label={tr('Entiti', 'Entity')}
   659→                  value={
   660→                    log.entity
   661→                      ? `${log.entity}${log.entityId ? ` #${log.entityId}` : ''}`
   662→                      : '—'
   663→                  }
   664→                  mono
   665→                />
   666→                <DetailField
   667→                  icon={Globe}
   668→                  label={tr('Alamat IP', 'IP Address')}
   669→                  value={log.ipAddress ?? '—'}
   670→                  mono
   671→                />
   672→              </div>
   673→
   674→              {/* User agent */}
   675→              {log.userAgent && (
   676→                <div className="rounded-lg bg-muted/40 border border-border/40 p-3">
   677→                  <div className="flex items-center gap-1.5 text-[10px] uppercase tracking-wider text-muted-foreground font-semibold mb-1">
   678→                    <Globe className="w-3 h-3" />
   679→                    {tr('User Agent', 'User Agent')}
   680→                  </div>
   681→                  <p className="text-[11px] font-mono text-muted-foreground break-all">{log.userAgent}</p>
   682→                </div>
   683→              )}
   684→
   685→              {/* Pretty-printed details */}
   686→              {prettyDetails && (
   687→                <div className="rounded-lg bg-zinc-900/95 border border-zinc-700/50 p-3 overflow-x-auto">
   688→                  <div className="flex items-center gap-1.5 text-[10px] uppercase tracking-wider text-teal-300 font-semibold mb-2">
   689→                    <FileText className="w-3 h-3" />
   690→                    {tr('Butiran (JSON)', 'Details (JSON)')}
   691→                  </div>
   692→                  <pre className="text-[11px] font-mono text-zinc-100 whitespace-pre-wrap break-all leading-relaxed">
   693→                    {prettyDetails}
   694→                  </pre>
   695→                </div>
   696→              )}
   697→
   698→              {!prettyDetails && !log.userAgent && (
   699→                <p className="text-xs text-muted-foreground italic text-center py-2">
   700→                  {tr('Tiada butiran tambahan.', 'No additional details.')}
   701→                </p>
   702→              )}
   703→            </div>
   704→          </motion.div>
   705→        )}
   706→      </AnimatePresence>
   707→    </div>
   708→  );
   709→}
