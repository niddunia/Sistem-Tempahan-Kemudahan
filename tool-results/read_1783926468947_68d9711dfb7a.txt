   118→   118→const initials = (name: string): string => {
   119→   119→  if (!name) return '?';
   120→   120→  const parts = name.trim().split(/\s+/);
   121→   121→  if (parts.length === 1) return parts[0].slice(0, 2).toUpperCase();
   122→   122→  return (parts[0][0] + parts[parts.length - 1][0]).toUpperCase();

   125→   125→const avatarColor = (name: string): string => {
   126→   126→  const palette = [
   127→   127→    'bg-teal-500/80 text-white',
   128→   128→    'bg-emerald-500/80 text-white',
   129→   129→    'bg-amber-500/80 text-white',
   130→   130→    'bg-rose-500/80 text-white',
   131→   131→    'bg-fuchsia-500/80 text-white',
   132→   132→    'bg-lime-500/80 text-white',
   133→   133→  ];
   134→   134→  let h = 0;
   135→   135→  for (let i = 0; i < name.length; i++) h = (h * 31 + name.charCodeAt(i)) >>> 0;
   136→   136→  return palette[h % palette.length];
   137→   137→};

   931→   931→      {/* Footer info */}

   942→   942→        </p>
   943→   943→      )}
   944→   944→    </motion.div>
   945→   945→  );
   946→   946→}
