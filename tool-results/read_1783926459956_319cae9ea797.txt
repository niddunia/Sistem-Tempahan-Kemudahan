     1→'use client';
     2→
     3→import { useState, useEffect, useMemo } from 'react';
     4→import { useT } from '@/hooks/use-t';

     6→import { GlassCard } from '@/components/glass-card';
     7→import { Button } from '@/components/ui/button';

    14→  SelectItem,
    15→  SelectTrigger,
    16→  SelectValue,
    17→} from '@/components/ui/select';
    18→import { Alert, AlertDescription, AlertTitle } from '@/components/ui/alert';
    19→import { Calendar } from '@/components/ui/calendar';
    20→import { motion } from 'framer-motion';
    21→import {

    46→/* ============================================================
    47→   BookingFormView — FR-01 Permohonan Tempahan
    48→   Sistem e-Tempahan PLTT-JTM
    49→   - Glassmorphism, teal/emerald/amber palette

    54→
    55→interface Facility {
    56→  id: string;
    57→  name: string;
    58→  category: 'COMPUTER_ROOM' | 'LECTURE_HALL' | string;
    59→  capacity: number;
    60→  location?: string | null;
    61→  equipment?: string | null;
    62→  operatingStart: string;
    63→  operatingEnd: string;
    64→  colorCode: string;
    65→  status: string;
    66→}
    67→
    68→interface ConflictAlternative {
    69→  startTime: string;
    70→  endTime: string;
    71→  facilityId: string;
    72→  facilityName: string;
    73→}
    74→
    75→interface ConflictResult {
    76→  hasConflict: boolean;
    77→  conflictingBookings: Array<{
    78→    id: string;
    79→    bookingRef: string;
    80→    startTime: string;
    81→    endTime: string;
    82→    status: string;
    83→    programName: string | null;
    84→  }>;
    85→  alternatives: ConflictAlternative[];
    86→}
    87→
    88→type CategoryFilter = 'all' | 'COMPUTER_ROOM' | 'LECTURE_HALL';
    89→
    90→export function BookingFormView() {
    91→  const { t, lang } = useT();
    92→  const { bookingPrefill, setBookingPrefill, setView } = useApp();
    93→
    94→  /* ---------------- form state ---------------- */
    95→  const [category, setCategory] = useState<CategoryFilter>('all');
    96→  const [facilityId, setFacilityId] = useState('');
    97→  const [date, setDate] = useState<Date | undefined>();
    98→  const [startTime, setStartTime] = useState('');
    99→  const [endTime, setEndTime] = useState('');
   100→  const [purpose, setPurpose] = useState('');
   101→  const [programName, setProgramName] = useState('');
   102→  const [participantCount, setParticipantCount] = useState('');
   103→  const [equipmentNeeded, setEquipmentNeeded] = useState('');
   104→  const [notes, setNotes] = useState('');
   105→
   106→  const [submitting, setSubmitting] = useState(false);
   107→  const [savingDraft, setSavingDraft] = useState(false);
   108→  const [checkingConflict, setCheckingConflict] = useState(false);
   109→  const [conflictResult, setConflictResult] = useState<ConflictResult | null>(null);
   110→  const [validated, setValidated] = useState(false);
   111→  const [prefillApplied, setPrefillApplied] = useState(false);
   112→
   113→  /* ---------------- fetch facilities ---------------- */
   114→  const { data: facilities, isLoading: loadingFacilities } = useQuery<Facility[]>({
   115→    queryKey: ['facilities'],
   116→    queryFn: async () => {
   117→      const r = await fetch('/api/facilities');
   118→      const j = await r.json();
   119→      return j.data as Facility[];
   120→    },
   121→  });
   122→
   123→  const filteredFacilities = useMemo(() => {
   124→    if (!facilities) return [];
   125→    if (category === 'all') return facilities;
   126→    return facilities.filter((f) => f.category === category);
   127→  }, [facilities, category]);
   128→
   129→  const selectedFacility = useMemo(
   130→    () => facilities?.find((f) => f.id === facilityId) ?? null,
   131→    [facilities, facilityId],
   132→  );
   133→
   134→  /* ---------------- pre-fill from AI parse ---------------- */
   135→  useEffect(() => {
   136→    if (prefillApplied || !bookingPrefill) return;
   137→    const p = bookingPrefill;
   138→    if (typeof p.facilityId === 'string' && p.facilityId) {
   139→      setFacilityId(p.facilityId);
   140→      const f = facilities?.find((x) => x.id === p.facilityId);
   141→      if (f) setCategory(f.category as CategoryFilter);
   142→    }
   143→    if (typeof p.date === 'string' && p.date) {
   144→      const d = new Date(p.date);
   145→      if (!isNaN(d.getTime())) {
   146→        d.setHours(0, 0, 0, 0);
   147→        setDate(d);
   148→      }
   149→    }
   150→    if (typeof p.startTime === 'string' && p.startTime) setStartTime(p.startTime);
   151→    if (typeof p.endTime === 'string' && p.endTime) setEndTime(p.endTime);
   152→    if (typeof p.participants === 'number' && p.participants > 0) {
   153→      setParticipantCount(String(p.participants));
   154→    }
   155→    if (typeof p.purpose === 'string' && p.purpose) setPurpose(p.purpose);
   156→    setPrefillApplied(true);
   157→    setBookingPrefill(null);
   158→    toast.info(
   159→      lang === 'bm'
   160→        ? 'Borang diisi automatik dari AI'
   161→        : 'Form auto-filled from AI',
   162→    );
   163→  }, [bookingPrefill, facilities, prefillApplied, setBookingPrefill, lang]);
   164→
   165→  /* ---------------- time options ---------------- */
   166→  const timeOptions = useMemo(() => {
   167→    const start = selectedFacility?.operatingStart ?? '08:00';
   168→    const end = selectedFacility?.operatingEnd ?? '22:00';
   169→    const startMin = parseInt(start.slice(0, 2), 10) * 60 + parseInt(start.slice(3, 5), 10);
   170→    const endMin = parseInt(end.slice(0, 2), 10) * 60 + parseInt(end.slice(3, 5), 10);
   171→    const opts: { value: string; label: string }[] = [];
   172→    for (let m = startMin; m < endMin; m += 30) {
   173→      const h = Math.floor(m / 60).toString().padStart(2, '0');
   174→      const min = (m % 60).toString().padStart(2, '0');
   175→      opts.push({ value: `${h}:${min}`, label: `${h}:${min}` });
   176→    }
   177→    return opts;
   178→  }, [selectedFacility]);
   179→
   180→  /* ---------------- derived: capacity + time validity ---------------- */
   181→  const participants = parseInt(participantCount, 10);
   182→  const capacityExceeded =
   183→    !!selectedFacility && !isNaN(participants) && participants > selectedFacility.capacity;
   184→  const invalidTime = !!startTime && !!endTime && startTime >= endTime;
   185→
   186→  /* ---------------- real-time conflict check (debounced) ---------------- */
   187→  useEffect(() => {
   188→    if (!facilityId || !date || !startTime || !endTime || invalidTime) {
   189→      setConflictResult(null);
   190→      setCheckingConflict(false);
   191→      return;
   192→    }
   193→    let cancelled = false;
   194→    setCheckingConflict(true);
   195→    const iso = new Date(date);
   196→    iso.setHours(0, 0, 0, 0);
   197→    const timer = setTimeout(async () => {
   198→      try {
   199→        const r = await fetch('/api/bookings/check-conflict', {
   200→          method: 'POST',
   201→          headers: { 'Content-Type': 'application/json' },
   202→          body: JSON.stringify({
   203→            facilityId,
   204→            eventDate: iso.toISOString(),
   205→            startTime,
   206→            endTime,
   207→          }),
   208→        });
   209→        const j = await r.json();
   210→        if (!cancelled && j?.data) setConflictResult(j.data as ConflictResult);
   211→      } catch {
   212→        /* network error — silent */
   213→      } finally {
   214→        if (!cancelled) setCheckingConflict(false);
   215→      }
   216→    }, 450);
   217→    return () => {
   218→      cancelled = true;
   219→      clearTimeout(timer);
   220→    };
   221→  }, [facilityId, date, startTime, endTime, invalidTime]);
   222→
   223→  /* ---------------- validation ---------------- */
   224→  const errors = useMemo<Record<string, string>>(() => {
   225→    const e: Record<string, string> = {};
   226→    if (!facilityId) e.facilityId = lang === 'bm' ? 'Sila pilih kemudahan' : 'Please select a facility';
   227→    if (!date) e.date = lang === 'bm' ? 'Sila pilih tarikh' : 'Please select a date';
   228→    if (!startTime) e.startTime = lang === 'bm' ? 'Sila pilih masa mula' : 'Please select start time';
   229→    if (!endTime) e.endTime = lang === 'bm' ? 'Sila pilih masa tamat' : 'Please select end time';
   230→    if (invalidTime) e.time = lang === 'bm' ? 'Masa tamat mesti selepas masa mula' : 'End time must be after start time';
   231→    if (!purpose.trim() || purpose.trim().length < 3)
   232→      e.purpose = lang === 'bm' ? 'Tujuan diperlukan (minimum 3 aksara)' : 'Purpose required (min 3 chars)';
   233→    if (!participantCount || isNaN(participants) || participants < 1)
   234→      e.participants = lang === 'bm' ? 'Bilangan peserta diperlukan' : 'Participant count required';
   235→    return e;
   236→  }, [facilityId, date, startTime, endTime, purpose, participantCount, participants, invalidTime, lang]);
   237→
   238→  const hasErrors = Object.keys(errors).length > 0;
   239→  const showErrors = validated;
   240→  const hasConflict = !!conflictResult?.hasConflict;
   241→  const canSubmit = !hasErrors && !capacityExceeded && !hasConflict && !checkingConflict;
   242→
   243→  /* ---------------- alternative slot click ---------------- */
   244→  const applyAlternative = (alt: ConflictAlternative) => {
   245→    setStartTime(alt.startTime);
   246→    setEndTime(alt.endTime);
   247→    toast.success(
   248→      lang === 'bm'
   249→        ? `Slot alternatif dipilih: ${alt.startTime} – ${alt.endTime}`
   250→        : `Alternative slot applied: ${alt.startTime} – ${alt.endTime}`,
   251→    );
   252→  };
   253→
   254→  /* ---------------- build payload helper ---------------- */
   255→  const buildPayload = (asDraft: boolean) => {
   256→    const iso = date ? new Date(date) : new Date();
   257→    iso.setHours(0, 0, 0, 0);
   258→    return {
   259→      facilityId,
   260→      eventDate: iso.toISOString(),
   261→      startTime: startTime || (selectedFacility?.operatingStart ?? '08:00'),
   262→      endTime: endTime || (selectedFacility?.operatingStart ?? '10:00'),
   263→      purpose: purpose.trim() || (lang === 'bm' ? 'Draf Permohonan' : 'Draft Booking'),
   264→      programName: programName.trim() || undefined,
   265→      participantCount: !isNaN(participants) && participants > 0 ? participants : 1,
   266→      equipmentNeeded: equipmentNeeded.trim() || undefined,
   267→      notes: notes.trim() || undefined,
   268→      isDraft: asDraft,
   269→    };
   270→  };
   271→
   272→  /* ---------------- submit handler ---------------- */
   273→  const handleSubmit = async (asDraft: boolean) => {
   274→    setValidated(true);
   275→    if (hasErrors) {
   276→      toast.error(lang === 'bm' ? 'Sila lengkapkan borang' : 'Please complete the form');
   277→      return;
   278→    }
   279→    if (asDraft) {
   280→      setSavingDraft(true);
   281→    } else {
   282→      if (capacityExceeded) {
   283→        toast.error(t('form_capacity_warn'));
   284→        return;
   285→      }
   286→      setSubmitting(true);
   287→      // Real-time conflict re-check before submission
   288→      try {
   289→        const iso = new Date(date!);
   290→        iso.setHours(0, 0, 0, 0);
   291→        const checkR = await fetch('/api/bookings/check-conflict', {
   292→          method: 'POST',
   293→          headers: { 'Content-Type': 'application/json' },
   294→          body: JSON.stringify({
   295→            facilityId,
   296→            eventDate: iso.toISOString(),
   297→            startTime,
   298→            endTime,
   299→          }),
   300→        });
   301→        const checkJ = await checkR.json();
   302→        if (checkJ?.data?.hasConflict) {
   303→          setConflictResult(checkJ.data as ConflictResult);
   304→          toast.error(t('form_conflict_warn'));
   305→          setSubmitting(false);
   306→          return;
   307→        }
   308→      } catch {
   309→        toast.error(lang === 'bm' ? 'Ralat semakan konflik' : 'Conflict check error');
   310→        setSubmitting(false);
   311→        return;
   312→      }
   313→    }
   314→
   315→    try {
   316→      const payload = buildPayload(asDraft);
   317→      const r = await fetch('/api/bookings', {
   318→        method: 'POST',
   319→        headers: { 'Content-Type': 'application/json' },
   320→        body: JSON.stringify(payload),
   321→      });
   322→      if (!r.ok) {
   323→        const j = await r.json().catch(() => null);
   324→        const msg = j?.message ?? (lang === 'bm' ? 'Permohonan gagal dihantar' : 'Submission failed');
   325→        toast.error(msg);
   326→        return;
   327→      }
   328→      if (asDraft) {
   329→        toast.info(lang === 'bm' ? 'Draf berjaya disimpan.' : 'Draft saved successfully.');
   330→      } else {
   331→        toast.success(t('form_success'));
   332→      }
   333→      setView('dashboard');
   334→    } catch {
   335→      toast.error(lang === 'bm' ? 'Ralat rangkaian. Cuba lagi.' : 'Network error. Try again.');
   336→    } finally {
   337→      setSubmitting(false);
   338→      setSavingDraft(false);
   339→    }
   340→  };
   341→
   342→  /* ---------------- helpers ---------------- */
   343→  const equipmentList = useMemo(() => {
   344→    if (!selectedFacility?.equipment) return [];
   345→    try {
   346→      const parsed = JSON.parse(selectedFacility.equipment);
   347→      return Array.isArray(parsed) ? parsed : [];
   348→    } catch {
   349→      return selectedFacility.equipment
   350→        .split(/[,;]/)
   351→        .map((s) => s.trim())
   352→        .filter(Boolean);
   353→    }
   354→  }, [selectedFacility]);
   355→
   356→  const locale = lang === 'bm' ? 'ms-MY' : 'en-GB';
   357→  const today = new Date();
   358→  today.setHours(0, 0, 0, 0);
   359→
   360→  const hasAlternatives = hasConflict && (conflictResult?.alternatives?.length ?? 0) > 0;
   361→
   362→  /* ---------------- render ---------------- */
   363→  return (
   364→    <motion.div
   365→      initial={{ opacity: 0, y: 10 }}
   366→      animate={{ opacity: 1, y: 0 }}
   367→      transition={{ duration: 0.3 }}
   368→      className="space-y-4 pb-8"
   369→    >
   370→      {/* ===== HERO ===== */}
   371→      <GlassCard className="p-5 lg:p-6 relative overflow-hidden">
   372→        <div className="absolute top-0 right-0 w-64 h-64 rounded-full bg-teal-500/10 blur-3xl pointer-events-none" />
   373→        <div className="absolute -bottom-12 -left-12 w-48 h-48 rounded-full bg-amber-500/10 blur-3xl pointer-events-none" />
   374→        <div className="relative z-10">
   375→          <div className="flex items-center gap-2 text-[11px] text-muted-foreground uppercase tracking-wider mb-2">
   376→            <Sparkles className="w-3 h-3 text-teal-500" />
   377→            <span>{lang === 'bm' ? 'Borang Permohonan' : 'Booking Application'}</span>
   378→          </div>
   379→          <h2 className="text-xl lg:text-2xl font-bold gradient-text mb-1">
   380→            {t('nav_book')}
   381→          </h2>
   382→          <p className="text-sm text-muted-foreground max-w-2xl">
   383→            {lang === 'bm'
   384→              ? 'Isi borang di bawah untuk memohon tempahan kemudahan. Semakan konflik & kapasiti dibuat secara automatik.'
   385→              : 'Complete the form below to apply for a facility booking. Conflict and capacity checks run automatically.'}
   386→          </p>
   387→        </div>
   388→      </GlassCard>
   389→
   390→      <div className="grid grid-cols-1 lg:grid-cols-3 gap-4 min-w-0">
   391→        {/* ===== MAIN FORM (2 cols) ===== */}
   392→        <GlassCard className="lg:col-span-2 p-5 lg:p-6 space-y-5 min-w-0 overflow-visible">
   393→          {/* Category + Facility */}
   394→          <div className="space-y-4">
   395→            <div className="space-y-2">
   396→              <Label className="text-xs font-semibold uppercase tracking-wider text-muted-foreground flex items-center gap-1.5">
   397→                <Tag className="w-3 h-3 text-teal-500" />
   398→                {lang === 'bm' ? 'Jenis Kemudahan' : 'Facility Type'}
   399→              </Label>
   400→              <div className="grid grid-cols-3 gap-2">
   401→                <CategoryChip
   402→                  active={category === 'all'}
   403→                  onClick={() => {
   404→                    setCategory('all');
   405→                    setFacilityId('');
   406→                  }}
   407→                  icon={<Building2 className="w-3.5 h-3.5" />}
   408→                  label={lang === 'bm' ? 'Semua' : 'All'}
   409→                />
   410→                <CategoryChip
   411→                  active={category === 'COMPUTER_ROOM'}
   412→                  onClick={() => {
   413→                    setCategory('COMPUTER_ROOM');
   414→                    setFacilityId('');
   415→                  }}
   416→                  icon={<Monitor className="w-3.5 h-3.5" />}
   417→                  label={lang === 'bm' ? 'Bilik Komputer' : 'Computer Room'}
   418→                />
   419→                <CategoryChip
   420→                  active={category === 'LECTURE_HALL'}
   421→                  onClick={() => {
   422→                    setCategory('LECTURE_HALL');
   423→                    setFacilityId('');
   424→                  }}
   425→                  icon={<GraduationCap className="w-3.5 h-3.5" />}
   426→                  label={lang === 'bm' ? 'Dewan Kuliah' : 'Lecture Hall'}
   427→                />
   428→              </div>
   429→            </div>
   430→
   431→            <div className="space-y-2">
   432→              <Label htmlFor="facility" className="text-xs font-semibold uppercase tracking-wider text-muted-foreground flex items-center gap-1.5">
   433→                <Building2 className="w-3 h-3 text-teal-500" />
   434→                {t('form_facility')} <span className="text-destructive">*</span>
   435→              </Label>
   436→              <Select value={facilityId} onValueChange={setFacilityId}>
   437→                <SelectTrigger
   438→                  id="facility"
   439→                  className={cn('glass-input w-full min-w-0', showErrors && errors.facilityId && 'border-destructive ring-destructive/30')}
   440→                >
   441→                  <SelectValue placeholder={
   442→                    loadingFacilities
   443→                      ? t('loading')
   444→                      : lang === 'bm'
   445→                        ? 'Pilih kemudahan…'
   446→                        : 'Select facility…'
   447→                  } />
   448→                </SelectTrigger>
   449→                <SelectContent className="glass-strong max-h-72">
   450→                  {filteredFacilities.length === 0 ? (
   451→                    <div className="px-3 py-2 text-xs text-muted-foreground">
   452→                      {lang === 'bm' ? 'Tiada kemudahan' : 'No facilities'}
   453→                    </div>
   454→                  ) : (
   455→                    filteredFacilities.map((f) => (
   456→                      <SelectItem key={f.id} value={f.id}>
   457→                        <span className="flex items-center gap-2 min-w-0 w-full">
   458→                          <span
   459→                            className="w-2 h-2 rounded-full shrink-0"
   460→                            style={{ background: f.colorCode }}
   461→                          />
   462→                          <span className="truncate">{f.name}</span>
   463→                          <span className="text-muted-foreground text-[10px] shrink-0">
   464→                            · {lang === 'bm' ? 'Kapasiti' : 'Cap.'} {f.capacity}
   465→                          </span>
   466→                        </span>
   467→                      </SelectItem>
   468→                    ))
   469→                  )}
   470→                </SelectContent>
   471→              </Select>
   472→              {showErrors && errors.facilityId && (
   473→                <p className="text-[11px] text-destructive flex items-center gap-1">
   474→                  <AlertCircle className="w-3 h-3" /> {errors.facilityId}
   475→                </p>
   476→              )}
   477→            </div>
   478→          </div>
   479→
   480→          {/* Date */}
   481→          <div className="space-y-2">
   482→            <Label className="text-xs font-semibold uppercase tracking-wider text-muted-foreground flex items-center gap-1.5">
   483→              <CalendarDays className="w-3 h-3 text-teal-500" />
   484→              {t('form_date')} <span className="text-destructive">*</span>
   485→            </Label>
   486→            <div className="flex flex-col sm:flex-row gap-3 items-start min-w-0 w-full overflow-visible">
   487→              <div className="glass-input rounded-lg p-1 inline-block overflow-visible shrink-0">
   488→                <Calendar
   489→                  mode="single"
   490→                  selected={date}
   491→                  onSelect={setDate}
   492→                  disabled={(d) => d < today}
   493→                  locale={locale === 'ms-MY' ? undefined : undefined}
   494→                  className="text-xs"
   495→                />
   496→              </div>
   497→              <div className="flex-1 min-w-0 space-y-2">
   498→                {date ? (
   499→                  <div className="glass-input rounded-md px-3 py-2.5 text-sm flex items-center gap-2">
   500→                    <CalendarDays className="w-4 h-4 text-teal-500" />
   501→                    <span className="font-medium">
   502→                      {date.toLocaleDateString(locale, {
   503→                        weekday: 'long',
   504→                        day: 'numeric',
   505→                        month: 'long',
   506→                        year: 'numeric',
   507→                      })}
   508→                    </span>
   509→                  </div>
   510→                ) : (
   511→                  <div className="glass-input rounded-md px-3 py-2.5 text-sm text-muted-foreground italic">
   512→                    {lang === 'bm' ? 'Pilih tarikh dari kalendar' : 'Pick a date from the calendar'}
   513→                  </div>
   514→                )}
   515→                {showErrors && errors.date && (
   516→                  <p className="text-[11px] text-destructive flex items-center gap-1">
   517→                    <AlertCircle className="w-3 h-3" /> {errors.date}
   518→                  </p>
   519→                )}
   520→              </div>
   521→            </div>
   522→          </div>
   523→
   524→          {/* Time */}
   525→          <div className="grid grid-cols-1 sm:grid-cols-2 gap-3">
   526→            <div className="space-y-2">
   527→              <Label className="text-xs font-semibold uppercase tracking-wider text-muted-foreground flex items-center gap-1.5">
